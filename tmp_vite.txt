   124	  const commandOutput = async (
   125	    executable: string,
   126	    args: string[],
   127	    label: string,
   128	  ): Promise<string> =>
   129	    await new Promise((resolve, reject) => {
   130	      const child = spawn(executable, args, {
   131	        stdio: ["ignore", "pipe", "pipe"],
   132	        windowsHide: true,
   133	      });
   134	      let stdout = "";
   135	      let stderr = "";
   136	      let settled = false;
   137	      child.stdout.on("data", (chunk) => (stdout += chunk));
   138	      child.stderr.on("data", (chunk) => (stderr += chunk));
   139	      child.on("error", (error) => {
   140	        if (settled) return;
   141	        settled = true;
   142	        reject(new Error(`Failed to start ${label}: ${error.message}`));
   143	      });
   144	      child.on("close", (code) => {
   145	        if (settled) return;
   146	        settled = true;
   147	        const output = stdout.trim();
   148	        const errorOutput = stderr.trim();
   149	        if (code !== 0) {
   150	          reject(
   151	            new Error(
   152	              `${label} failed with exit code ${code ?? -1}${
   153	                errorOutput || output ? `: ${errorOutput || output}` : ""
   154	              }`,
   155	            ),
   156	          );
   157	          return;
   158	        }
   159	        resolve(output);
   160	      });
   161	    });

   331	  const MAX_CHAT_HISTORY_TURNS = 12;
   332	  const MAX_CHAT_TURN_LENGTH = 2_000;
   333	  const MAX_CHAT_HISTORY_CHARS = 12_000;
   334	
   335	  type ChatTurn = { role: "user" | "assistant"; content: string };
   336	
   337	  // Mirrors normalize_chat_history in src-tauri/src/commands/codex_assistant.rs.
   338	  const normalizeChatHistory = (history: unknown): ChatTurn[] => {
   339	    if (history === undefined || history === null) return [];
   340	    if (!Array.isArray(history)) throw new Error("对话历史格式不正确");
   341	    let normalized: ChatTurn[] = [];
   342	    for (const entry of history) {
   343	      if (!entry || typeof entry !== "object")
   344	        throw new Error("对话历史格式不正确");
   345	      const { role, content } = entry as { role?: unknown; content?: unknown };
   346	      if (role !== "user" && role !== "assistant")
   347	        throw new Error("对话历史包含非法角色");
   348	      if (typeof content !== "string") throw new Error("对话历史格式不正确");
   349	      const trimmed = content.trim();
   350	      if (!trimmed) continue;
   351	      normalized.push({
   352	        role,
   353	        content: Array.from(trimmed).slice(0, MAX_CHAT_TURN_LENGTH).join(""),
   354	      });
   355	    }
   356	    if (normalized.length > MAX_CHAT_HISTORY_TURNS) {
   357	      normalized = normalized.slice(-MAX_CHAT_HISTORY_TURNS);
   358	    }
   359	    while (
   360	      normalized.reduce((total, turn) => total + turn.content.length, 0) >
   361	      MAX_CHAT_HISTORY_CHARS
   362	    ) {
   363	      normalized.shift();
   364	    }
   365	    return normalized;
   366	  };

   387	    const raw = body.decision ?? body.approved ?? body.action;
   388	    if (
   389	      raw === true ||
   390	      raw === "accept" ||
   391	      raw === "acceptForSession" ||
   392	      raw === "approved" ||
   393	      raw === "approve"

   399	    if (
   400	      raw === false ||
   401	      raw === "decline" ||
   402	      raw === "denied" ||
   403	      raw === "deny"
   404	    ) {
   405	      return "decline" as PreviewApprovalDecision;
   406	    }
   407	    if (raw === "cancel") return "cancel" as PreviewApprovalDecision;
   408	    throw new Error("请选择允许、拒绝或取消任务");

   450	  const probeToolVersion = async (
   451	    executable: string,
   452	    cwd?: string,

   455	    return await new Promise<{ success: boolean; output: string }>(

   458	          cwd,

   463	        let text = "";
   464	        child.stdout.on("data", (chunk) => (text += chunk));
   465	        child.stderr.on("data", (chunk) => (text += chunk));
   466	        child.on("error", () => resolve({ success: false, output: "" }));
   467	        child.on("close", (code) =>
   468	          resolve({
   469	            success: code === 0 && Boolean(text.trim()),
   470	            output: text.trim(),
   471	          }),
   472	        );
   473	      },

   477	  return {
   478	    name: "cc-switch-web-codex-assistant",
   479	    configureServer(server: {
   480	      middlewares: {
   481	        use: (
   482	          handler: (
   483	            request: IncomingMessage,
   484	            response: ServerResponse,
   485	            next: () => void,
   486	          ) => void,
   487	        ) => void;
   488	      };
   489	    }) {
   490	      server.middlewares.use((request, response, next) => {
   491	        if (!request.url?.startsWith(WEB_ASSISTANT_BASE)) return next();
   492	        const requestUrl = new URL(request.url, "http://127.0.0.1");
   493	        const route = requestUrl.pathname;
   494	        if (
   495	          request.method === "GET" &&
   496	          route === `${WEB_ASSISTANT_BASE}/events`
   497	        ) {
   498	          response.writeHead(200, {
   499	            "Content-Type": "text/event-stream",
   500	            "Cache-Control": "no-cache",
   501	            Connection: "keep-alive",
   502	          });
   503	          response.write("retry: 1000\n\n");
   504	          listeners.add(response);
   505	          request.on("close", () => listeners.delete(response));
   506	          return;
   507	        }
   508	        if (
   509	          request.method === "GET" &&
   510	          route === `${WEB_ASSISTANT_BASE}/recent-events`
   511	        ) {
   512	          return sendJson(response, 200, { events: recentEvents });
   513	        }
   514	        if (
   515	          request.method === "GET" &&
   516	          route === `${WEB_ASSISTANT_BASE}/status`
   517	        ) {
   518	          void (async () => {
   519	            const result = await probeToolVersion("codex");
   520	            sendJson(response, 200, {
   521	              available: result.success,
   522	              version: result.output || null,
   523	            });
   524	          })();
   525	          return;
   526	        }
   527	        if (
   528	          request.method === "GET" &&
   529	          route === `${WEB_ASSISTANT_BASE}/desktop-status`
   530	        ) {
   531	          void detectDesktopApp(DESKTOP_APPS["codex-desktop"])
   532	            .then((status) =>
   533	              sendJson(response, 200, {
   534	                installed: status.installed,
   535	                version: status.version,
   536	                path: status.path,
   537	              }),
   538	            )
   539	            .catch((error) =>
   540	              sendJson(response, 400, {
   541	                error: error instanceof Error ? error.message : String(error),
   542	              }),
   543	            );
   544	          return;
   545	        }
   546	        if (
   547	          request.method === "GET" &&
   548	          route === `${WEB_ASSISTANT_BASE}/desktop-app-status`
   549	        ) {
   550	          void Promise.resolve()
   551	            .then(() =>
   552	              detectDesktopApp(
   553	                desktopManifest(requestUrl.searchParams.get("app")),
   554	              ),
   555	            )
   556	            .then((status) => sendJson(response, 200, status))
   557	            .catch((error) =>
   558	              sendJson(response, 400, {
   559	                error: error instanceof Error ? error.message : String(error),
   560	              }),
   561	            );
   562	          return;
   563	        }
   564	        if (request.method !== "POST")
   565	          return sendJson(response, 405, { error: "Method not allowed" });
   566	        void readJson(request)
   567	          .then(async (body) => {
   568	            if (
   569	              route === `${WEB_ASSISTANT_BASE}/session` ||
   570	              route === `${WEB_ASSISTANT_BASE}/sessions`
   571	            ) {
   572	              const session = createSession(body.history);
   573	              emit({ sessionId: session.id, kind: "session", status: "ready" });
   574	              return sendJson(response, 201, { sessionId: session.id });
   575	            }
   576	            if (
   577	              route === `${WEB_ASSISTANT_BASE}/chat` ||
   578	              route === `${WEB_ASSISTANT_BASE}/message` ||
   579	              route === `${WEB_ASSISTANT_BASE}/messages`
   580	            ) {
   581	              const requestText =
   582	                typeof body.request === "string"
   583	                  ? body.request
   584	                  : typeof body.message === "string"
   585	                    ? body.message
   586	                    : typeof body.content === "string"
   587	                      ? body.content
   588	                      : "";
   589	              if (!requestText.trim()) throw new Error("请输入想问的内容");
   590	              if (requestText.length > MAX_REQUEST_LENGTH)
   591	                throw new Error("请求过长");
   592	
   593	              const session = resolveSession(body);
   594	              const requestContent = requestText.trim();
   595	              const suppliedHistory =
   596	                body.history === undefined
   597	                  ? session.history
   598	                  : normalizeChatHistory(body.history);
   599	              const history = normalizeChatHistory(suppliedHistory);
   600	
   601	              if (looksLikeOperationRequest(requestContent)) {
   602	                const previewMessage =
   603	                  "这是浏览器预览。我可以展示操作确认，但不会安装软件、修改配置或执行真实命令。";
   604	                session.history = normalizeChatHistory([
   605	                  ...history,
   606	                  { role: "user", content: requestContent },
   607	                  { role: "assistant", content: previewMessage },
   608	                ]);
   609	                emitPreviewApproval(session, requestContent);
   610	                return sendJson(response, 202, {
   611	                  accepted: true,
   612	                  sessionId: session.id,
   613	                });
   614	              }
   615	
   616	              const answer =
   617	                "这是浏览器开发预览中的模拟回答。我可以在同一个对话里帮助你了解 AI Agent、安装方式、Provider 和配置流程；涉及操作时会先展示确认卡，而且预览不会执行真实命令或修改系统。";
   618	              session.history = normalizeChatHistory([
   619	                ...history,
   620	                { role: "user", content: requestContent },
   621	                { role: "assistant", content: answer },
   622	              ]);
   623	              emit({ sessionId: session.id, kind: "started" });
   624	              emit({
   625	                sessionId: session.id,
   626	                kind: "message",
   627	                message: answer,
   628	              });
   629	              emit({
   630	                sessionId: session.id,
   631	                kind: "finished",
   632	                success: true,
   633	                cancelled: false,
   634	                message: answer,
   635	              });
   636	              return sendJson(response, 202, {
   637	                accepted: true,
   638	                sessionId: session.id,
   639	              });
   640	            }
   641	            if (
   642	              route === `${WEB_ASSISTANT_BASE}/approval` ||
   643	              route === `${WEB_ASSISTANT_BASE}/approvals`
   644	            ) {
   645	              const session = getSession(body.sessionId ?? body.session_id);
   646	              if (!session) throw new Error("Codex 对话不存在或已失效");
   647	              const approvalId = body.approvalId ?? body.approval_id;
   648	              if (
   649	                typeof approvalId !== "string" ||
   650	                approvalId !== session.pendingApprovalId
   651	              ) {
   652	                throw new Error("确认请求不存在或已失效");
   653	              }
   654	              const decision = approvalDecision(body);
   655	              delete session.pendingApprovalId;
   656	              const message =
   657	                decision === "accept" || decision === "acceptForSession"
   658	                  ? "已记录允许。浏览器预览不会执行真实操作。"
   659	                  : decision === "cancel"
   660	                    ? "已取消任务。浏览器预览未执行任何更改。"
   661	                    : "已拒绝操作。浏览器预览未执行任何更改。";
   662	              session.history = normalizeChatHistory([
   663	                ...session.history,
   664	                { role: "assistant", content: message },
   665	              ]);
   666	              emit({
   667	                sessionId: session.id,
   668	                kind: "message",
   669	                message,
   670	              });
   671	              emit({
   672	                sessionId: session.id,
   673	                kind: "finished",
   674	                success: true,
   675	                cancelled: decision === "cancel",
   676	                message,
   677	              });
   678	              return sendJson(response, 200, {
   679	                accepted: true,
   680	                sessionId: session.id,
   681	                approvalId,
   682	                decision,
   683	                simulated: true,
   684	                message,
   685	              });
   686	            }
   687	            if (route === `${WEB_ASSISTANT_BASE}/cancel`) {
   688	              const session = getSession(
   689	                body.sessionId ?? body.session_id ?? body.runId,
   690	              );
   691	              if (!session || !session.pendingApprovalId) {
   692	                throw new Error("没有正在运行的 Codex 任务");
   693	              }
   694	              delete session.pendingApprovalId;
   695	              emit({
   696	                sessionId: session.id,
   697	                kind: "finished",
   698	                success: false,
   699	                cancelled: true,
   700	                message: "已取消任务。浏览器预览未执行任何更改。",
   701	              });
   702	              return sendJson(response, 200, { cancelled: true });
   703	            }
   704	            if (route === `${WEB_ASSISTANT_BASE}/close`) {
   705	              const sessionId = body.sessionId ?? body.session_id;
   706	              if (typeof sessionId !== "string")
   707	                throw new Error("Codex 对话不存在或已失效");
   708	              const closed = sessions.delete(sessionId);
   709	              return sendJson(response, 200, { closed });
   710	            }
   711	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-check-updates`) {
   712	              const status = await checkDesktopAppUpdates(
   713	                desktopManifest(body.app),
   714	              );
   715	              return sendJson(response, 200, status);
   716	            }
   717	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-action`) {
   718	              const manifest = desktopManifest(body.app);
   719	              if (
   720	                body.action !== "install" &&
   721	                body.action !== "update" &&
   722	                body.action !== "uninstall"
   723	              ) {
   724	                throw new Error(
   725	                  `Unsupported desktop lifecycle action: ${String(body.action ?? "")}`,
   726	                );
   727	              }
   728	              const status = await detectDesktopApp(manifest);
   729	              return sendJson(response, 200, {
   730	                ...status,
   731	                reason: `${manifest.displayName} ${body.action} is not executed in the browser preview`,
   732	              });
   733	            }
   734	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-launch`) {
   735	              await launchDesktopApp(desktopManifest(body.app));
   736	              return sendJson(response, 200, { launched: true });
   737	            }
   738	            if (route === `${WEB_ASSISTANT_BASE}/launch-desktop`) {
   739	              await launchDesktopApp(DESKTOP_APPS["codex-desktop"]);
   740	              return sendJson(response, 200, { launched: true });
   741	            }
   742	            sendJson(response, 404, { error: "Unknown bridge endpoint" });
   743	          })
   744	          .catch((error) =>
   745	            sendJson(response, 400, {
   746	              error: error instanceof Error ? error.message : String(error),
   747	            }),
   748	          );
   749	      });
   750	    },
   751	  };
   752	}
