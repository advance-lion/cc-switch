   117	  const commandOutput = async (
   118	    executable: string,
   119	    args: string[],
   120	    label: string,
   121	  ): Promise<string> =>
   122	    await new Promise((resolve, reject) => {
   123	      const child = spawn(executable, args, {
   124	        stdio: ["ignore", "pipe", "pipe"],
   125	        windowsHide: true,
   126	      });
   127	      let stdout = "";
   128	      let stderr = "";
   129	      let settled = false;
   130	      child.stdout.on("data", (chunk) => (stdout += chunk));
   131	      child.stderr.on("data", (chunk) => (stderr += chunk));
   132	      child.on("error", (error) => {
   133	        if (settled) return;
   134	        settled = true;
   135	        reject(new Error(`Failed to start ${label}: ${error.message}`));
   136	      });
   137	      child.on("close", (code) => {
   138	        if (settled) return;
   139	        settled = true;
   140	        const output = stdout.trim();
   141	        const errorOutput = stderr.trim();
   142	        if (code !== 0) {
   143	          reject(
   144	            new Error(
   145	              `${label} failed with exit code ${code ?? -1}${
   146	                errorOutput || output ? `: ${errorOutput || output}` : ""
   147	              }`,
   148	            ),
   149	          );
   150	          return;
   151	        }
   152	        resolve(output);
   153	      });
   154	    });

   324	  const MAX_CHAT_HISTORY_TURNS = 12;
   325	  const MAX_CHAT_TURN_LENGTH = 2_000;
   326	  const MAX_CHAT_HISTORY_CHARS = 12_000;

   328	  type ChatTurn = { role: "user" | "assistant"; content: string };

   330	  // Mirrors normalize_chat_history in src-tauri/src/commands/codex_assistant.rs.
   331	  const normalizeChatHistory = (history: unknown): ChatTurn[] => {
   332	    if (history === undefined || history === null) return [];
   333	    if (!Array.isArray(history)) throw new Error("对话历史格式不正确");
   334	    let normalized: ChatTurn[] = [];
   335	    for (const entry of history) {
   336	      if (!entry || typeof entry !== "object")
   337	        throw new Error("对话历史格式不正确");
   338	      const { role, content } = entry as { role?: unknown; content?: unknown };
   339	      if (role !== "user" && role !== "assistant")
   340	        throw new Error("对话历史包含非法角色");
   341	      if (typeof content !== "string") throw new Error("对话历史格式不正确");
   342	      const trimmed = content.trim();
   343	      if (!trimmed) continue;
   344	      normalized.push({
   345	        role,
   346	        content: Array.from(trimmed).slice(0, MAX_CHAT_TURN_LENGTH).join(""),
   347	      });
   348	    }
   349	    if (normalized.length > MAX_CHAT_HISTORY_TURNS) {
   350	      normalized = normalized.slice(-MAX_CHAT_HISTORY_TURNS);
   351	    }
   352	    while (
   353	      normalized.reduce((total, turn) => total + turn.content.length, 0) >
   354	      MAX_CHAT_HISTORY_CHARS
   355	    ) {
   356	      normalized.shift();
   357	    }
   358	    return normalized;
   359	  };

   542	              });

   544	                sessionId: session.id,
   545	                kind: "finished",
   546	                success: true,
   547	                cancelled: false,

   549	              });

   551	                accepted: true,
   552	                sessionId: session.id,
   553	              });

   562	              if (

   587	                kind: "finished",

   612	                success: false,
   613	                cancelled: true,
   614	                message: "已取消任务。浏览器预览未执行任何更改。",
   615	              });
   616	              return sendJson(response, 200, { cancelled: true });
   617	            }
   618	            if (route === `${WEB_ASSISTANT_BASE}/close`) {
   619	              const sessionId = body.sessionId ?? body.session_id;
   620	              if (typeof sessionId !== "string")
   621	                throw new Error("Codex 对话不存在或已失效");
   622	              const closed = sessions.delete(sessionId);
   623	              return sendJson(response, 200, { closed });
   624	            }
   625	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-check-updates`) {
   626	              const status = await checkDesktopAppUpdates(
   627	                desktopManifest(body.app),
   628	              );
   629	              return sendJson(response, 200, status);
   630	            }
   631	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-action`) {
   632	              const manifest = desktopManifest(body.app);
   633	              if (
   634	                body.action !== "install" &&
   635	                body.action !== "update" &&
   636	                body.action !== "uninstall"
   637	              ) {
   638	                throw new Error(
   639	                  `Unsupported desktop lifecycle action: ${String(body.action ?? "")}`,
   640	                );
   641	              }
   642	              const status = await detectDesktopApp(manifest);
   643	              return sendJson(response, 200, {
   644	                ...status,
   645	                reason: `${manifest.displayName} ${body.action} is not executed in the browser preview`,
   646	              });
   647	            }
   648	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-launch`) {
   649	              await launchDesktopApp(desktopManifest(body.app));
   650	              return sendJson(response, 200, { launched: true });
   651	            }
   652	            if (route === `${WEB_ASSISTANT_BASE}/launch-desktop`) {
   653	              await launchDesktopApp(DESKTOP_APPS["codex-desktop"]);
   654	              return sendJson(response, 200, { launched: true });
   655	            }
   656	            sendJson(response, 404, { error: "Unknown bridge endpoint" });
   657	          })
   658	          .catch((error) =>
   659	            sendJson(response, 400, {
   660	              error: error instanceof Error ? error.message : String(error),
   661	            }),
   662	          );
   663	      });
   664	    },
   665	  };
   666	}
