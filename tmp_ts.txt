   197	async function webAssistantRequest<T>(
   198	  path: string,
   199	  body?: Record<string, unknown>,
   200	): Promise<T> {
   201	  const response = await fetch(`${WEB_ASSISTANT_BRIDGE_BASE}${path}`, {
   202	    method: body ? "POST" : "GET",
   203	    headers: body ? { "Content-Type": "application/json" } : undefined,
   204	    body: body ? JSON.stringify(body) : undefined,
   205	  });
   206	  const payload = (await response.json()) as T & { error?: string };
   207	  if (!response.ok)
   208	    throw new Error(payload.error || "Web assistant bridge failed");
   209	  return payload;
   210	}

   577	    tools?: string[],
   578	    wslShellByTool?: Record<
   579	      string,
   580	      { wslShell?: string | null; wslShellFlag?: string | null }

   586	      latest_version: string | null;

   590	      wsl_distro: string | null;
   591	    }>
   592	  > {
   593	    if (isCodexAssistantWebBridgeActive() && tools?.length === 1) {
   594	      const tool = tools[0];
   595	      const status =
   596	        tool === "codex"
   597	          ? await webAssistantRequest<{
   598	              available: boolean;
   599	              version: string | null;
   600	            }>("/status")
   601	          : { available: false, version: null };
   602	      return [
   603	        {
   604	          name: tool,
   605	          version: status.available ? status.version : null,
   606	          latest_version: null,
   607	          error: status.available ? null : "Codex CLI unavailable",
   608	          installed_but_broken: false,
   609	          env_type: "windows",
   610	          wsl_distro: null,
   611	        },
   612	      ];
   613	    }
   614	    return await invoke("get_tool_versions", { tools, wslShellByTool });
