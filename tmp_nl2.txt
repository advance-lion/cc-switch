   115	fn validate_input(input: &str) -> Result<String, String> {
   116	    let input = input.trim();
   117	    if input.is_empty() {
   118	        return Err("请输入消息".to_string());
   119	    }
   120	    if input.chars().count() > MAX_INPUT_LENGTH {
   121	        return Err(format!("消息过长（最多 {MAX_INPUT_LENGTH} 个字符）"));
   122	    }
   123	    Ok(input.to_string())
   124	}

   126	fn resolve_workspace_dir(raw: Option<&str>) -> Result<PathBuf, String> {
   127	    let path = match raw.map(str::trim).filter(|value| !value.is_empty()) {
   128	        Some(raw) => fs::canonicalize(raw).map_err(|error| format!("无法访问工作目录: {error}"))?,
   129	        None => {
   130	            let path = crate::config::get_app_config_dir()
   131	                .join("codex-assistant")
   132	                .join("workspace");
   133	            fs::create_dir_all(&path).map_err(|error| format!("无法创建助手工作目录: {error}"))?;
   134	            fs::canonicalize(path).map_err(|error| format!("无法访问助手工作目录: {error}"))?
   135	        }
   136	    };
   137	    if !path.is_dir() {
   138	        return Err("工作目录必须是文件夹".to_string());
   139	    }
   140	    Ok(path)
   141	}

   143	/// Windows npm/pnpm installations expose Codex as a `.cmd` shim. Never pass
   144	/// assistant input through cmd.exe; resolve the fixed official JS entry and
   145	/// invoke Node with an argv array instead.
   146	#[cfg(target_os = "windows")]
   147	fn codex_assistant_command_for(executable: &Path) -> Result<Command, String> {
   148	    let is_batch = executable
   149	        .extension()
   150	        .and_then(|extension| extension.to_str())
   151	        .is_some_and(|extension| {
   152	            extension.eq_ignore_ascii_case("cmd") || extension.eq_ignore_ascii_case("bat")
   153	        });
   154	    if !is_batch {
   155	        return Ok(Command::new(executable));
   156	    }
   157	
   158	    let bin_dir = executable
   159	        .parent()
   160	        .ok_or_else(|| "Codex CLI 批处理入口没有父目录".to_string())?;
   161	    let entry = [
   162	        bin_dir.join("node_modules/@openai/codex/bin/codex.js"),
   163	        bin_dir.join("../@openai/codex/bin/codex.js"),
   164	    ]
   165	    .into_iter()
   166	    .find(|candidate| candidate.is_file())
   167	    .ok_or_else(|| {
   168	        format!(
   169	            "无法从 Codex CLI 入口 {} 定位官方 Node 启动文件",
   170	            executable.display()
   171	        )
   172	    })?;
   173	    let entry = super::misc::windows_shell_compatible_path(&entry);
   174	    let local_node = bin_dir.join("node.exe");
   175	    let node = if local_node.is_file() {
   176	        super::misc::windows_shell_compatible_path(&local_node)
   177	    } else {
   178	        let node = super::misc::resolve_tool_executable_for_gui("node");
   179	        super::misc::windows_shell_compatible_path(&node)
   180	    };
   181	    let mut command = Command::new(node);
   182	    command.arg(entry);
   183	    Ok(command)
   184	}

   188	    Ok(Command::new(executable))
   189	}

   191	fn codex_assistant_home() -> PathBuf {
   192	    let configured = crate::codex_config::get_codex_config_dir();
   193	    if !crate::config::is_test_sandbox()
   194	        || crate::settings::get_codex_override_dir().is_some()
   195	        || configured.join("config.toml").is_file()
   196	        || configured.join("auth.json").is_file()
   197	    {
   198	        return configured;
   199	    }
   200	
   201	    #[cfg(debug_assertions)]
   202	    if let Some(real_home) = dirs::home_dir() {
   203	        let host_codex = real_home.join(".codex");
   204	        if host_codex.join("config.toml").is_file() || host_codex.join("auth.json").is_file() {
   205	            return host_codex;
   206	        }
   207	    }
   208	
   209	    configured
   210	}

   212	fn request_key(id: &Value) -> Result<String, String> {
   213	    match id {
   214	        Value::String(_) | Value::Number(_) => {
   215	            serde_json::to_string(id).map_err(|error| format!("无效的 JSON-RPC 请求 ID: {error}"))
   216	        }
   217	        _ => Err("JSON-RPC 请求 ID 必须是字符串或数字".to_string()),
   218	    }
   219	}

   221	fn response_error(value: &Value) -> String {
   222	    value
   223	        .pointer("/error/message")
   224	        .and_then(Value::as_str)
   225	        .unwrap_or("Codex app-server 请求失败")
   226	        .to_string()

   229	impl CodexAppServerSession {
   230	    fn write_message(&self, message: &Value) -> Result<(), String> {
   231	        let mut stdin = self
   232	            .stdin
   233	            .lock()
   234	            .map_err(|_| "Codex app-server 输入锁不可用".to_string())?;
   235	        serde_json::to_writer(&mut *stdin, message)
   236	            .map_err(|error| format!("无法编码 Codex app-server 消息: {error}"))?;
   237	        stdin
   238	            .write_all(b"\n")
   239	            .and_then(|_| stdin.flush())
   240	            .map_err(|error| format!("无法写入 Codex app-server: {error}"))
   241	    }
   242	
   243	    fn notify(&self, method: &str, params: Option<Value>) -> Result<(), String> {
   244	        let mut message = json!({ "method": method });
   245	        if let Some(params) = params {
   246	            message["params"] = params;
   247	        }
   248	        self.write_message(&message)
   249	    }
   250	
   251	    fn request(&self, method: &str, params: Value) -> Result<Value, String> {
   252	        let id = Value::from(self.next_request_id.fetch_add(1, Ordering::Relaxed));
   253	        let key = request_key(&id)?;
   254	        let (sender, receiver) = mpsc::channel();
   255	        self.pending_responses
   256	            .lock()
   257	            .map_err(|_| "Codex app-server 响应锁不可用".to_string())?
   258	            .insert(key.clone(), sender);
   259	
   260	        if let Err(error) = self.write_message(&json!({
   261	            "id": id,
   262	            "method": method,
   263	            "params": params,
   264	        })) {
   265	            if let Ok(mut pending) = self.pending_responses.lock() {
   266	                pending.remove(&key);
   267	            }
   268	            return Err(error);
   269	        }
   270	
   271	        match receiver.recv_timeout(RPC_TIMEOUT) {
   272	            Ok(result) => result,
   273	            Err(mpsc::RecvTimeoutError::Timeout) => {
   274	                if let Ok(mut pending) = self.pending_responses.lock() {
   275	                    pending.remove(&key);
   276	                }
   277	                Err(format!("Codex app-server 请求超时: {method}"))
   278	            }
   279	            Err(mpsc::RecvTimeoutError::Disconnected) => {
   280	                Err("Codex app-server 已断开连接".to_string())
   281	            }
   282	        }
   283	    }
   284	
   285	    fn fail_pending_responses(&self, error: &str) {
   286	        if let Ok(mut pending) = self.pending_responses.lock() {
   287	            for (_, sender) in pending.drain() {
   288	                let _ = sender.send(Err(error.to_string()));
   289	            }
   290	        }
   291	    }
   292	}

   311	/// Reject any server-initiated request with a JSON-RPC "unsupported" error.
   312	/// With `approvalPolicy: "never"` the server never sends approval requests,
   313	/// so this path is defensive only.
   314	fn handle_server_request(session: &Arc<CodexAppServerSession>, message: &Value) {
   315	    let Some(request_id) = message.get("id").cloned() else {
   316	        return;
   317	    };
   318	    if request_key(&request_id).is_err() {
   319	        return;
   320	    }
   321	    let _ = session.write_message(&json!({
   322	        "id": request_id,
   323	        "error": { "code": -32601, "message": "Unsupported app-server request" }
   324	    }));
   325	}

   327	fn handle_protocol_message(session: &Arc<CodexAppServerSession>, message: Value) {
   328	    if let Some(id) = message.get("id") {
   329	        if message.get("method").is_some() {
   330	            handle_server_request(session, &message);
   331	            return;
   332	        }
   333	        if let Ok(key) = request_key(id) {
   334	            let sender = session
   335	                .pending_responses
   336	                .lock()
   337	                .ok()
   338	                .and_then(|mut pending| pending.remove(&key));
   339	            if let Some(sender) = sender {
   340	                let result = if message.get("error").is_some() {
   341	                    Err(response_error(&message))
   342	                } else {
   343	                    Ok(message.get("result").cloned().unwrap_or(Value::Null))
   344	                };
   345	                let _ = sender.send(result);
   346	            }
   347	        }
   348	        return;
   349	    }
   350	
   351	    let Some(method) = message.get("method").and_then(Value::as_str) else {
   352	        return;
   353	    };
   354	    let params = message.get("params").cloned().unwrap_or(Value::Null);
   355	    match method {
   356	        "item/agentMessage/delta" => {
   357	            if let Some(delta) = params.get("delta").and_then(Value::as_str) {
   358	                emit_event(
   359	                    session.events.as_ref(),
   360	                    CodexAssistantEvent::Message {
   361	                        session_id: session.id.clone(),
   362	                        message: delta.to_string(),
   363	                    },
   364	                );
   365	            }
   366	        }
   367	        "item/commandExecution/outputDelta" | "item/commandExecution/terminalInteraction" => {
   368	            if let Some(delta) = params.get("delta").and_then(Value::as_str) {
   369	                emit_event(
   370	                    session.events.as_ref(),
   371	                    CodexAssistantEvent::Log {
   372	                        session_id: session.id.clone(),
   373	                        message: delta.to_string(),
   374	                    },
   375	                );
   376	            }
   377	        }
   378	        "item/fileChange/outputDelta" | "item/fileChange/patchUpdated" => {
   379	            emit_event(
   380	                session.events.as_ref(),
   381	                CodexAssistantEvent::Log {
   382	                    session_id: session.id.clone(),
   383	                    message: params.to_string(),
   384	                },
   385	            );
   386	        }
   387	        "turn/started" => {
   388	            session.turn_active.store(true, Ordering::Release);
   389	            if let Some(turn_id) = params
   390	                .get("turnId")
   391	                .or_else(|| params.pointer("/turn/id"))
   392	                .and_then(Value::as_str)
   393	            {
   394	                if let Ok(mut active) = session.active_turn_id.lock() {
   395	                    *active = Some(turn_id.to_string());
   396	                }
   397	            }
   398	            emit_event(
   399	                session.events.as_ref(),
   400	                CodexAssistantEvent::Started {
   401	                    session_id: session.id.clone(),
   402	                },
   403	            );
   404	        }
   405	        "turn/completed" => {
   406	            session.turn_active.store(false, Ordering::Release);
   407	            if let Ok(mut active) = session.active_turn_id.lock() {
   408	                *active = None;
   409	            }
   410	            let status = params.pointer("/turn/status").and_then(Value::as_str);
   411	            let success = matches!(status, Some("completed"));
   412	            let cancelled = matches!(status, Some("interrupted" | "cancelled"));
   413	            let error = params
   414	                .pointer("/turn/error/message")
   415	                .or_else(|| params.pointer("/turn/error"))
   416	                .and_then(Value::as_str)
   417	                .map(str::to_string)
   418	                .or_else(|| {
   419	                    (!success && !cancelled).then(|| status.unwrap_or("Codex 回合失败").to_string())
   420	                });
   421	            emit_event(
   422	                session.events.as_ref(),
   423	                CodexAssistantEvent::Finished {
   424	                    session_id: session.id.clone(),
   425	                    success,
   426	                    cancelled,
   427	                    message: error,
   428	                },
   429	            );
   430	        }
   431	        "error" => {
   432	            let error = params
   433	                .pointer("/error/message")
   434	                .and_then(Value::as_str)
   435	                .unwrap_or("Codex app-server 报告错误")
   436	                .to_string();
   437	            emit_event(
   438	                session.events.as_ref(),
   439	                CodexAssistantEvent::Stderr {
   440	                    session_id: session.id.clone(),
   441	                    message: error,
   442	                },
   443	            );
   444	        }
   445	        _ => {}
   446	    }
   447	}

   449	fn start_protocol_readers(
   450	    session: Arc<CodexAppServerSession>,
   451	    stdout: impl std::io::Read + Send + 'static,
   452	    stderr: impl std::io::Read + Send + 'static,
   453	) {
   454	    let stdout_session = session.clone();
   455	    thread::spawn(move || {
   456	        for line in BufReader::new(stdout).lines() {
   457	            match line {
   458	                Ok(line) if !line.trim().is_empty() => match serde_json::from_str::<Value>(&line) {
   459	                    Ok(message) => handle_protocol_message(&stdout_session, message),
   460	                    Err(error) => emit_event(
   461	                        stdout_session.events.as_ref(),
   462	                        CodexAssistantEvent::Stderr {
   463	                            session_id: stdout_session.id.clone(),
   464	                            message: format!("无法解析 Codex app-server 消息: {error}"),
   465	                        },
   466	                    ),
   467	                },
   468	                Ok(_) => {}
   469	                Err(error) => {
   470	                    emit_event(
   471	                        stdout_session.events.as_ref(),
   472	                        CodexAssistantEvent::Stderr {
   473	                            session_id: stdout_session.id.clone(),
   474	                            message: format!("读取 Codex app-server 输出失败: {error}"),
   475	                        },
   476	                    );
   477	                    break;
   478	                }
   479	            }
   480	        }
   481	        stdout_session.fail_pending_responses("Codex app-server 已断开连接");
   482	    });
   483	
   484	    let stderr_session = session.clone();
   485	    thread::spawn(move || {
   486	        for line in BufReader::new(stderr).lines().map_while(Result::ok) {
   487	            emit_event(
   488	                stderr_session.events.as_ref(),
   489	                CodexAssistantEvent::Stderr {
   490	                    session_id: stderr_session.id.clone(),
   491	                    message: line,
   492	                },
   493	            );
   494	        }
   495	    });
   496	
   497	    thread::spawn(move || {
   498	        loop {
   499	            let status = match session.child.lock() {
   500	                Ok(mut child) => child.try_wait(),
   501	                Err(_) => return,
   502	            };
   503	            match status {
   504	                Ok(Some(_)) => break,
   505	                Ok(None) => thread::sleep(Duration::from_millis(100)),
   506	                Err(_) => break,
   507	            }
   508	        }
   509	        let turn_was_active = session.turn_active.swap(false, Ordering::AcqRel);
   510	        session.fail_pending_responses("Codex app-server 已退出");
   511	        if !session.closing.load(Ordering::Acquire) {
   512	            let message = if turn_was_active {
   513	                "Codex app-server 在回复过程中意外退出"
   514	            } else {
   515	                "Codex app-server 已意外断开，请重新发送消息以建立新会话"
   516	            };
   517	            log::error!(
   518	                "Codex assistant session {} disconnected: {message}",
   519	                session.id
   520	            );
   521	            emit_event(
   522	                session.events.as_ref(),
   523	                CodexAssistantEvent::Disconnected {
   524	                    session_id: session.id.clone(),
   525	                    message: message.to_string(),
   526	                },
   527	            );
   528	        }
   529	        if let Ok(mut sessions) = SESSIONS.lock() {
   530	            sessions.remove(&session.id);
   531	        }
   532	    });
   533	}

   545	fn spawn_app_server(app: AppHandle) -> Result<Arc<CodexAppServerSession>, String> {
   546	    let executable = super::misc::resolve_tool_executable_for_gui("codex");
   547	    let mut command = codex_assistant_command_for(&executable)?;
   548	    let codex_home = codex_assistant_home();
   549	    fs::create_dir_all(&codex_home).map_err(|error| format!("无法创建 Codex 配置目录: {error}"))?;
   550	    configure_app_server_command(&mut command, &codex_home);
   551	    #[cfg(target_os = "windows")]
   552	    {
   553	        use std::os::windows::process::CommandExt;
   554	        command.creation_flags(0x08000000);
   555	    }
   556	
   557	    let mut child = command
   558	        .spawn()
   559	        .map_err(|error| format!("无法启动 Codex app-server: {error}"))?;
   560	    let stdin = child
   561	        .stdin
   562	        .take()
   563	        .ok_or_else(|| "无法打开 Codex app-server 输入".to_string())?;
   564	    let stdout = child
   565	        .stdout
   566	        .take()
   567	        .ok_or_else(|| "无法读取 Codex app-server 输出".to_string())?;
   568	    let stderr = child
   569	        .stderr

   654	pub fn send_codex_assistant_message(session_id: String, message: String) -> Result<(), String> {
   655	    let input = validate_input(&message)?;
   656	    let session = get_session(&session_id)?;
   657	    if session
   658	        .turn_active
   659	        .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
   660	        .is_err()
   661	    {
   662	        return Err("当前 Codex 回合尚未结束".to_string());
   663	    }
   664	    let result = (|| {
   665	        let thread_id = session
   666	            .thread_id
   667	            .lock()
   668	            .map_err(|_| "Codex thread 状态锁不可用".to_string())?
   669	            .clone()
   670	            .ok_or_else(|| "Codex thread 尚未初始化".to_string())?;
   671	        let result = session.request(
   672	            "turn/start",
   673	            json!({
   674	                "threadId": thread_id,
   675	                "input": [{ "type": "text", "text": input }]
   676	            }),
   677	        )?;
   678	        let turn_id = extract_id(&result, "/turn/id", "turn id")?;
   679	        if session.turn_active.load(Ordering::Acquire) {
   680	            *session
   681	                .active_turn_id
   682	                .lock()
   683	                .map_err(|_| "Codex turn 状态锁不可用".to_string())? = Some(turn_id);
   684	        }
   685	        Ok(())
   686	    })();
   687	    if result.is_err() {
   688	        session.turn_active.store(false, Ordering::Release);
   689	    }
   690	    result
   691	}

   693	#[tauri::command]
   694	pub fn respond_codex_assistant_approval(
   695	    session_id: String,
   696	    approval_id: String,
   697	    decision: String,
   698	) -> Result<bool, String> {
   699	    let session = get_session(&session_id)?;
   700	    let decision = decision.trim();
   701	    let pending = {
   702	        let mut approvals = session
   703	            .pending_approvals
   704	            .lock()
   705	            .map_err(|_| "Codex 审批状态锁不可用".to_string())?;
   706	        let pending = approvals
   707	            .get(&approval_id)
   708	            .cloned()
   709	            .ok_or_else(|| "审批请求不存在或已处理".to_string())?;
   710	        if !pending.available_decisions.contains(decision) {
   711	            return Err(format!("当前审批不支持 decision: {decision}"));
   712	        }
   713	        if decision == "acceptForSession"
   714	            && pending.method == COMMAND_APPROVAL_METHOD
   715	            && pending.high_risk
   716	        {
   717	            return Err("高风险命令不能在本会话中持续授权".to_string());
   718	        }
   719	        approvals.remove(&approval_id);
   720	        pending
   721	    };
   722	    if let Err(error) = session.write_message(&json!({
   723	        "id": pending.request_id,
   724	        "result": { "decision": decision }
   725	    })) {
   726	        if let Ok(mut approvals) = session.pending_approvals.lock() {
   727	            approvals.insert(approval_id, pending);
   728	        }
   729	        return Err(error);
   730	    }
   731	    Ok(true)
   732	}
