    47	const clampFloatingPosition = (
    48	  position: FloatingPosition,
    49	): FloatingPosition => ({
    50	  x: Math.min(
    51	    Math.max(FLOATING_BUTTON_MARGIN, position.x),
    52	    Math.max(
    53	      FLOATING_BUTTON_MARGIN,
    54	      window.innerWidth - FLOATING_BUTTON_SIZE - FLOATING_BUTTON_MARGIN,
    55	    ),
    56	  ),
    57	  y: Math.min(
    58	    Math.max(FLOATING_BUTTON_MARGIN, position.y),
    59	    Math.max(
    60	      FLOATING_BUTTON_MARGIN,
    61	      window.innerHeight - FLOATING_BUTTON_SIZE - FLOATING_BUTTON_MARGIN,
    62	    ),
    63	  ),
    64	});

    67	  clampFloatingPosition({

    89	function ApprovalCard({
    90	  approval,
    91	  responding,
    92	  onRespond,
    93	}: {
    94	  approval: CodexAssistantApproval;
    95	  responding: boolean;
    96	  onRespond: (decision: CodexAssistantApprovalDecision) => void;
    97	}) {
    98	  const { t } = useTranslation();
    99	  const isCommand = approval.type === "command";
   100	  const supports = (decision: CodexAssistantApprovalDecision) =>
   101	    approval.availableDecisions.includes(decision);
   102	
   103	  return (
   104	    <div
   105	      className="space-y-3 rounded-xl border border-amber-500/30 bg-amber-500/[0.055] p-3"
   106	      data-testid={`codex-approval-${approval.id}`}
   107	    >
   108	      <div className="flex items-start gap-2.5">
   109	        <span className="mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-lg bg-amber-500/15 text-amber-700 dark:text-amber-200">
   110	          {isCommand ? (
   111	            <Terminal className="h-3.5 w-3.5" />
   112	          ) : (
   113	            <FilePenLine className="h-3.5 w-3.5" />
   114	          )}
   115	        </span>
   116	        <div className="min-w-0">
   117	          <p className="text-sm font-semibold">
   118	            {isCommand
   119	              ? t("codexAssistant.commandApproval", {
   120	                  defaultValue: "Command approval",
   121	                })
   122	              : t("codexAssistant.fileApproval", {
   123	                  defaultValue: "File change approval",
   124	                })}
   125	          </p>
   126	          {approval.reason && (
   127	            <p className="mt-0.5 text-xs leading-relaxed text-muted-foreground">
   128	              {approval.reason}
   129	            </p>
   130	          )}
   131	        </div>
   132	      </div>
   133	
   134	      {approval.command && (
   135	        <pre className="max-h-36 overflow-auto whitespace-pre-wrap break-all rounded-lg border border-border/70 bg-background/70 px-2.5 py-2 font-mono text-[11px] leading-relaxed">
   136	          {approval.command}
   137	        </pre>
   138	      )}
   139	
   140	      <dl className="space-y-1 text-[11px] text-muted-foreground">
   141	        {approval.cwd && (
   142	          <div className="flex gap-2">
   143	            <dt className="shrink-0 font-medium text-foreground/80">
   144	              {t("codexAssistant.workingDirectory", {
   145	                defaultValue: "Working directory",
   146	              })}
   147	            </dt>
   148	            <dd className="min-w-0 break-all">{approval.cwd}</dd>
   149	          </div>
   150	        )}
   151	        {approval.networkHost && (
   152	          <div className="flex gap-2">
   153	            <dt className="shrink-0 font-medium text-foreground/80">
   154	              {t("codexAssistant.networkHost", { defaultValue: "Network" })}
   155	            </dt>
   156	            <dd className="min-w-0 break-all">{approval.networkHost}</dd>
   157	          </div>
   158	        )}
   159	        {approval.grantRoot && (
   160	          <div className="flex gap-2">
   161	            <dt className="shrink-0 font-medium text-foreground/80">
   162	              {t("codexAssistant.grantRoot", { defaultValue: "Write access" })}
   163	            </dt>
   164	            <dd className="min-w-0 break-all">{approval.grantRoot}</dd>
   165	          </div>
   166	        )}
   167	      </dl>
   168	
   169	      <div className="flex flex-wrap justify-end gap-2">
   170	        {supports("cancel") && (
   171	          <Button
   172	            type="button"
   173	            variant="outline"
   174	            size="sm"
   175	            disabled={responding}
   176	            onClick={() => onRespond("cancel")}
   177	          >
   178	            {t("codexAssistant.cancelApproval", {
   179	              defaultValue: "Cancel task",
   180	            })}
   181	          </Button>
   182	        )}
   183	        {supports("decline") && (
   184	          <Button
   185	            type="button"
   186	            variant="outline"
   187	            size="sm"
   188	            disabled={responding}
   189	            onClick={() => onRespond("decline")}
   190	          >
   191	            {t("codexAssistant.declineApproval", { defaultValue: "Deny" })}
   192	          </Button>
   193	        )}
   194	        {supports("accept") && (
   195	          <Button
   196	            type="button"
   197	            variant="outline"
   198	            size="sm"
   199	            disabled={responding}
   200	            onClick={() => onRespond("accept")}
   201	          >
   202	            {t("codexAssistant.approveOnce", { defaultValue: "Allow once" })}
   203	          </Button>
   204	        )}
   205	        {approval.allowForSession && supports("acceptForSession") && (
   206	          <Button
   207	            type="button"
   208	            size="sm"
   209	            disabled={responding}
   210	            onClick={() => onRespond("acceptForSession")}
   211	          >
   212	            {responding && <Loader2 className="h-3.5 w-3.5 animate-spin" />}
   213	            {t("codexAssistant.approveForSession", {
   214	              defaultValue: "Allow for session",
   215	            })}
   216	          </Button>
   217	        )}
   218	      </div>
   219	    </div>
   220	  );
   221	}

   223	/** A persistent Codex app-server conversation with explicit approval prompts. */
   224	export function CodexAssistantDock({
   225	  providerReady,
   226	  onOpenCodexConfiguration,
   227	  onDismiss,
   228	  onInstallationCompleted,
   229	  installIntent,
   230	  openRequestId = 0,
   231	}: CodexAssistantDockProps) {
   232	  const { t } = useTranslation();
   233	  const webBridgeActive = isCodexAssistantWebBridgeActive();
   234	  const [open, setOpen] = useState(false);
   235	  const [showFloatingGreeting, setShowFloatingGreeting] = useState(true);
   236	  const [chatMessages, setChatMessages] = useState<CodexAssistantChatTurn[]>(
   237	    [],
   238	  );
   239	  const [runtime, setRuntime] = useState<CodexRuntime | null>(null);
   240	  const [checking, setChecking] = useState(false);
   241	  const [installing, setInstalling] = useState(false);
   242	  const [request, setRequest] = useState("");
   243	  const [logs, setLogs] = useState<string[]>([]);
   244	  const [approvals, setApprovals] = useState<CodexAssistantApproval[]>([]);
   245	  const [respondingApprovalIds, setRespondingApprovalIds] = useState<
   246	    Set<string>
   247	  >(new Set());
   248	  const [runError, setRunError] = useState<string | null>(null);
   249	  const [startingSession, setStartingSession] = useState(false);
   250	  const [running, setRunning] = useState(false);
   251	  const [stopping, setStopping] = useState(false);
   252	  const [sessionId, setSessionId] = useState<string | null>(null);
   253	  const [floatingPosition, setFloatingPosition] =
   254	    useState<FloatingPosition | null>(null);
   255	  const floatingDockRef = useRef<HTMLDivElement>(null);
   256	  const latestFloatingPositionRef = useRef<FloatingPosition | null>(null);
   257	  const pendingFloatingPositionRef = useRef<FloatingPosition | null>(null);
   258	  const dragAnimationFrameRef = useRef<number | null>(null);
   259	  const dragStateRef = useRef<{
   260	    pointerId: number;
   261	    startClientX: number;
   262	    startClientY: number;
   263	    startPosition: FloatingPosition;
   264	    moved: boolean;
   265	  } | null>(null);
   266	  const suppressOpenRef = useRef(false);
   267	  const sessionIdRef = useRef<string | null>(null);
   268	  const streamAssistantMessageRef = useRef(false);
   269	  const lastHandledOpenRequestRef = useRef(0);
   270	
   271	  const appendLog = useCallback((line: string) => {
   272	    setLogs((current) => [...current, line].slice(-MAX_VISIBLE_LOG_LINES));
   273	  }, []);
   274	
   275	  useEffect(() => {
   276	    const storedPosition = localStorage.getItem(FLOATING_POSITION_STORAGE_KEY);
   277	    if (storedPosition) {
   278	      try {
   279	        const parsed = JSON.parse(storedPosition) as Partial<FloatingPosition>;
   280	        if (typeof parsed.x === "number" && typeof parsed.y === "number") {
   281	          const next = clampFloatingPosition(parsed as FloatingPosition);
   282	          latestFloatingPositionRef.current = next;
   283	          setFloatingPosition(next);
   284	          return;
   285	        }
   286	      } catch {
   287	        // Ignore stale stored coordinates.
   288	      }
   289	    }
   290	    const next = defaultFloatingPosition();
   291	    latestFloatingPositionRef.current = next;
   292	    setFloatingPosition(next);
   293	  }, []);
   294	
   295	  useEffect(() => {
   296	    if (
   297	      openRequestId <= 0 ||
   298	      openRequestId === lastHandledOpenRequestRef.current
   299	    ) {
   300	      return;
   301	    }
   302	    lastHandledOpenRequestRef.current = openRequestId;
   303	    if (installIntent) {
   304	      setRequest(
   305	        t("codexAssistant.installRequest", { app: installIntent.appName }),
   306	      );
   307	    }
   308	    setShowFloatingGreeting(false);
   309	    setOpen(true);
   310	  }, [installIntent, openRequestId, t]);
   311	
   312	  useEffect(() => {
   313	    const keepFloatingButtonVisible = () => {
   314	      const next = clampFloatingPosition(
   315	        latestFloatingPositionRef.current ?? defaultFloatingPosition(),
   316	      );
   317	      latestFloatingPositionRef.current = next;
   318	      setFloatingPosition(next);
   319	    };
   320	    window.addEventListener("resize", keepFloatingButtonVisible);
   321	    return () =>
   322	      window.removeEventListener("resize", keepFloatingButtonVisible);
   323	  }, []);
   324	
   325	  useEffect(
   326	    () => () => {
   327	      if (dragAnimationFrameRef.current !== null) {
   328	        cancelAnimationFrame(dragAnimationFrameRef.current);
   329	      }
   330	      const activeSessionId = sessionIdRef.current;
   331	      sessionIdRef.current = null;
   332	      if (activeSessionId) {
   333	        void settingsApi.closeCodexAssistantSession(activeSessionId);
   334	      }
   335	    },
   336	    [],
   337	  );
   338	
   339	  const refreshRuntime = useCallback(async () => {
   340	    setChecking(true);
   341	    try {
   342	      const [next] = await settingsApi.getToolVersions(["codex"]);
   343	      setRuntime(next ?? null);
   344	    } catch (error) {
   345	      toast.error(t("codexAssistant.runtimeCheckFailed"), {
   346	        description: extractErrorMessage(error) || undefined,
   347	      });
   348	    } finally {
   349	      setChecking(false);
   350	    }
   351	  }, [t]);
   352	
   353	  useEffect(() => {
   354	    let dispose = false;
   355	    let unlisten: (() => void) | undefined;
   356	
   357	    void settingsApi
   358	      .onCodexAssistantEvent((event) => {
   359	        if (event.sessionId !== sessionIdRef.current) return;
   360	
   361	        switch (event.kind) {
   362	          case "started":
   363	            streamAssistantMessageRef.current = false;
   364	            setRunning(true);
   365	            return;
   366	          case "log":
   367	            if (event.message) appendLog(event.message);
   368	            return;
   369	          case "stderr":
   370	            if (event.message) appendLog(`stderr · ${event.message}`);
   371	            return;
   372	          case "disconnected":
   373	            sessionIdRef.current = null;
   374	            setSessionId(null);
   375	            streamAssistantMessageRef.current = false;
   376	            setRunning(false);
   377	            setStopping(false);
   378	            setApprovals([]);
   379	            setRespondingApprovalIds(new Set());
   380	            setRunError(event.message);
   381	            toast.error(t("codexAssistant.runFailed"), {
   382	              description: event.message,
   383	            });
   384	            return;
   385	          case "message":
   386	            if (event.message) {
   387	              setChatMessages((current) => {
   388	                if (streamAssistantMessageRef.current) {
   389	                  const last = current.at(-1);
   390	                  if (last?.role === "assistant") {
   391	                    return [
   392	                      ...current.slice(0, -1),
   393	                      { ...last, content: last.content + event.message },
   394	                    ];
   395	                  }
   396	                }
   397	                streamAssistantMessageRef.current = true;
   398	                return [
   399	                  ...current,
   400	                  { role: "assistant", content: event.message },
   401	                ];
   402	              });
   403	            }
   404	            return;
   405	          case "approval":
   406	            setApprovals((current) => [
   407	              ...current.filter((item) => item.id !== event.approval.id),
   408	              event.approval,
   409	            ]);
   410	            return;
   411	          case "finished":
   412	            streamAssistantMessageRef.current = false;
   413	            setRunning(false);
   414	            setStopping(false);
   415	            setApprovals([]);
   416	            setRespondingApprovalIds(new Set());
   417	            if (!event.success && !event.cancelled) {
   418	              const message = event.message || t("codexAssistant.runFailed");
   419	              setRunError(message);
   420	              toast.error(t("codexAssistant.runFailed"), {
   421	                description: message,
   422	              });
   423	            }
   424	        }
   425	      })
   426	      .then((listener) => {
   427	        if (dispose) listener();
   428	        else unlisten = listener;
   429	      });
   430	
   431	    return () => {
   432	      dispose = true;
   433	      unlisten?.();
   434	    };
   435	  }, [appendLog, t]);
   436	
   437	  useEffect(() => {
   438	    if (open) void refreshRuntime();
   439	  }, [open, refreshRuntime]);
   440	
   441	  const installCodex = async () => {
   442	    setInstalling(true);
   443	    try {
   444	      await settingsApi.runToolLifecycleAction(["codex"], "install");
   445	      await refreshRuntime();
   446	      onInstallationCompleted?.("codex");
   447	      toast.success(t("codexAssistant.runtimeInstalled"));
   448	    } catch (error) {
   449	      toast.error(t("codexAssistant.runtimeInstallFailed"), {
   450	        description: extractErrorMessage(error) || undefined,
   451	      });
   452	    } finally {
   453	      setInstalling(false);
   454	    }
   455	  };
   456	
   457	  const cliReady = Boolean(runtime?.version && !runtime?.installed_but_broken);
   458	  const effectiveCliReady = cliReady || webBridgeActive;
   459	  const effectiveProviderReady = providerReady || webBridgeActive;
   460	  const ready = effectiveCliReady && effectiveProviderReady;
   461	  const blockedReason = !effectiveCliReady
   462	    ? t("codexAssistant.cliRequired")
   463	    : !effectiveProviderReady
   464	      ? t("codexAssistant.providerRequired")
   465	      : null;
   466	
   467	  const ensureSession = async (): Promise<string> => {
   468	    if (sessionIdRef.current) return sessionIdRef.current;
   469	    setStartingSession(true);
   470	    try {
   471	      const nextSessionId = await settingsApi.startCodexAssistantSession();
   472	      sessionIdRef.current = nextSessionId;
   473	      setSessionId(nextSessionId);
   474	      return nextSessionId;
   475	    } finally {
   476	      setStartingSession(false);
   477	    }
   478	  };
   479	
   480	  const sendMessage = async () => {
   481	    const text = request.trim();
   482	    if (!text || running || startingSession) return;
   483	
   484	    setRequest("");
   485	    setRunError(null);
   486	    setLogs([]);
   487	    streamAssistantMessageRef.current = false;
   488	    setChatMessages((current) => [...current, { role: "user", content: text }]);
   489	    try {
   490	      const activeSessionId = await ensureSession();
   491	      setRunning(true);
   492	      await settingsApi.sendCodexAssistantMessage(activeSessionId, text);
   493	    } catch (error) {
   494	      setRunning(false);
   495	      const message =
   496	        extractErrorMessage(error) || t("codexAssistant.runFailed");
   497	      setRunError(message);
   498	      toast.error(t("codexAssistant.runFailed"), { description: message });
   499	    }
   500	  };
   501	
   502	  const respondToApproval = async (
   503	    approvalId: string,
   504	    decision: CodexAssistantApprovalDecision,
   505	  ) => {
   506	    const activeSessionId = sessionIdRef.current;
   507	    if (!activeSessionId || respondingApprovalIds.has(approvalId)) return;
   508	    setRespondingApprovalIds((current) => new Set(current).add(approvalId));
   509	    try {
   510	      await settingsApi.respondCodexAssistantApproval(
   511	        activeSessionId,
   512	        approvalId,
   513	        decision,
   514	      );
   515	      setApprovals((current) =>
   516	        current.filter((approval) => approval.id !== approvalId),
   517	      );
   518	    } catch (error) {
   519	      toast.error(
   520	        t("codexAssistant.approvalFailed", {
   521	          defaultValue: "Could not send the approval decision",
   522	        }),
   523	        { description: extractErrorMessage(error) || undefined },
   524	      );
   525	    } finally {
   526	      setRespondingApprovalIds((current) => {
   527	        const next = new Set(current);
   528	        next.delete(approvalId);
   529	        return next;
   530	      });
   531	    }
   532	  };
   533	
   534	  const cancelRun = async () => {
   535	    const activeSessionId = sessionIdRef.current;
   536	    if (!activeSessionId || stopping) return;
   537	    setStopping(true);
   538	    try {
   539	      await settingsApi.cancelCodexAssistantRun(activeSessionId);
   540	      appendLog(t("codexAssistant.cancelling"));
   541	    } catch (error) {
   542	      setStopping(false);
   543	      toast.error(t("codexAssistant.cancelFailed"), {
   544	        description: extractErrorMessage(error) || undefined,
   545	      });
   546	    }
   547	  };
   548	
   549	  const closeSession = () => {
   550	    const activeSessionId = sessionIdRef.current;
   551	    sessionIdRef.current = null;
   552	    setSessionId(null);
   553	    streamAssistantMessageRef.current = false;
   554	    setRunning(false);
   555	    setStopping(false);
   556	    setApprovals([]);
   557	    setRespondingApprovalIds(new Set());
   558	    if (activeSessionId) {
   559	      void settingsApi
   560	        .closeCodexAssistantSession(activeSessionId)
   561	        .catch(() => undefined);
   562	    }
   563	  };
   564	
   565	  const persistFloatingPosition = (position: FloatingPosition) => {
   566	    localStorage.setItem(
   567	      FLOATING_POSITION_STORAGE_KEY,
   568	      JSON.stringify(position),
   569	    );
   570	  };
   571	
   572	  const moveFloatingButton = (position: FloatingPosition) => {
   573	    latestFloatingPositionRef.current = position;
   574	    pendingFloatingPositionRef.current = position;
   575	    if (dragAnimationFrameRef.current !== null) return;
   576	
   577	    dragAnimationFrameRef.current = requestAnimationFrame(() => {
   578	      const next = pendingFloatingPositionRef.current;
   579	      const dock = floatingDockRef.current;
   580	      if (next && dock) {
   581	        dock.style.transform = `translate3d(${next.x}px, ${next.y}px, 0)`;
   582	      }
   583	      dragAnimationFrameRef.current = null;
   584	    });
   585	  };
   586	
   587	  const handlePointerDown = (event: ReactPointerEvent<HTMLButtonElement>) => {
   588	    if (event.button !== 0) return;
   589	    event.preventDefault();
   590	    const current =
   591	      latestFloatingPositionRef.current ??
   592	      floatingPosition ??
   593	      defaultFloatingPosition();
   594	    event.currentTarget.setPointerCapture(event.pointerId);
   595	    dragStateRef.current = {
   596	      pointerId: event.pointerId,
   597	      startClientX: event.clientX,
   598	      startClientY: event.clientY,
   599	      startPosition: current,
   600	      moved: false,
   601	    };
   602	  };
   603	
   604	  const handlePointerMove = (event: ReactPointerEvent<HTMLButtonElement>) => {
   605	    const drag = dragStateRef.current;
   606	    if (!drag || drag.pointerId !== event.pointerId) return;
   607	    event.preventDefault();
   608	    const deltaX = event.clientX - drag.startClientX;
   609	    const deltaY = event.clientY - drag.startClientY;
   610	    if (Math.abs(deltaX) + Math.abs(deltaY) > 4) drag.moved = true;
   611	    if (!drag.moved) return;
   612	    moveFloatingButton(
   613	      clampFloatingPosition({
   614	        x: drag.startPosition.x + deltaX,
   615	        y: drag.startPosition.y + deltaY,
   616	      }),
   617	    );
   618	  };
   619	
   620	  const finishDrag = (event: ReactPointerEvent<HTMLButtonElement>) => {
   621	    const drag = dragStateRef.current;
   622	    if (!drag || drag.pointerId !== event.pointerId) return;
   623	    dragStateRef.current = null;
   624	    if (!drag.moved) return;
   625	    const nextPosition = clampFloatingPosition({
   626	      x: drag.startPosition.x + event.clientX - drag.startClientX,
   627	      y: drag.startPosition.y + event.clientY - drag.startClientY,
   628	    });
   629	    moveFloatingButton(nextPosition);
   630	    setFloatingPosition(nextPosition);
   631	    persistFloatingPosition(nextPosition);
   632	    suppressOpenRef.current = true;
   633	  };
   634	
   635	  const openAssistant = () => {
   636	    if (suppressOpenRef.current) {
   637	      suppressOpenRef.current = false;
   638	      return;
   639	    }
   640	    setShowFloatingGreeting(false);
   641	    setOpen(true);
   642	  };
   643	
   644	  const closeAssistant = () => {
   645	    setOpen(false);
   646	    closeSession();
   647	    onDismiss?.();
   648	  };
   649	
   650	  useEffect(() => {
   651	    if (!open) return;
   652	    const handleKeyDown = (event: KeyboardEvent) => {
   653	      if (event.key === "Escape") {
   654	        event.preventDefault();
   655	        closeAssistant();
   656	      }
   657	    };
   658	    window.addEventListener("keydown", handleKeyDown);
   659	    return () => window.removeEventListener("keydown", handleKeyDown);
   660	  }, [open, onDismiss]);
   661	
   662	  return (
   663	    <>
   664	      <div
   665	        ref={floatingDockRef}
   666	        style={
   667	          floatingPosition
   668	            ? {
   669	                left: 0,
   670	                top: 0,
   671	                transform: `translate3d(${floatingPosition.x}px, ${floatingPosition.y}px, 0)`,
   672	              }
   673	            : { right: FLOATING_BUTTON_MARGIN, top: "calc(50% - 17px)" }
   674	        }
   675	        className="fixed z-[120] isolate will-change-transform"
   676	      >
   677	        {showFloatingGreeting && (
   678	          <div className="pointer-events-none absolute right-[calc(100%+10px)] top-1/2 hidden w-56 -translate-y-1/2 rounded-2xl border border-violet-500/25 bg-background/75 py-2 pl-3 pr-8 text-left text-xs leading-relaxed text-foreground/75 opacity-90 shadow-[0_8px_24px_-16px_rgba(76,29,149,0.45)] backdrop-blur-[2px] sm:block">
   679	            <span>{t("codexAssistant.floatingGreeting")}</span>
   680	            <button
   681	              type="button"
   682	              className="pointer-events-auto absolute right-1.5 top-1.5 flex h-5 w-5 items-center justify-center rounded-full text-muted-foreground/70 transition-colors hover:bg-violet-500/10 hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-500/40"
   683	              onClick={() => setShowFloatingGreeting(false)}
   684	              aria-label={t("codexAssistant.dismissGreeting")}
   685	              title={t("codexAssistant.dismissGreeting")}
   686	            >
   687	              <X className="h-3 w-3" />
   688	            </button>
   689	          </div>
   690	        )}
   691	        <button
   692	          type="button"
   693	          onClick={openAssistant}
   694	          onPointerDown={handlePointerDown}
   695	          onPointerMove={handlePointerMove}
   696	          onPointerUp={finishDrag}
   697	          onPointerCancel={finishDrag}
   698	          className="flex h-[34px] w-[34px] cursor-grab touch-none select-none items-center justify-center rounded-full border border-violet-500/35 bg-background text-violet-600 shadow-[0_10px_22px_-10px_rgba(109,40,217,0.7)] backdrop-blur transition-[background-color,box-shadow] hover:bg-violet-500/10 active:cursor-grabbing dark:text-violet-300"
   699	          aria-label={t("codexAssistant.open")}
   700	          title={t("codexAssistant.open")}
   701	        >
   702	          <CodexIcon size={19} className="dark:invert-0" />
   703	        </button>
   704	      </div>
   705	
   706	      <aside
   707	        aria-hidden={!open}
   708	        className={cn(
   709	          "fixed inset-y-0 right-0 z-[130] flex w-[390px] max-w-[calc(100vw-24px)] flex-col border-l border-border bg-background shadow-[-18px_0_44px_-24px_rgba(15,23,42,0.45)] transition-transform duration-300 ease-out",
   710	          open ? "translate-x-0" : "translate-x-full",
   711	        )}
   712	      >
   713	        <header className="flex items-center justify-between border-b border-border px-5 py-4">
   714	          <div className="flex items-center gap-3">
   715	            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-gradient-to-br from-violet-500 to-indigo-500 text-white shadow-sm">
   716	              <CodexIcon size={17} className="dark:invert-0" />
   717	            </span>
   718	            <div>
   719	              <h2 className="text-sm font-semibold">
   720	                {t("codexAssistant.title")}
   721	              </h2>
   722	              <p className="text-xs text-muted-foreground">
   723	                {t("codexAssistant.subtitle")}
   724	              </p>
   725	            </div>
   726	          </div>
   727	          <Button
   728	            variant="ghost"
   729	            size="icon"
   730	            className="h-8 w-8"
   731	            onClick={closeAssistant}
   732	            aria-label={t("common.close")}
   733	          >
   734	            <X className="h-4 w-4" />
   735	          </Button>
   736	        </header>
   737	
   738	        <div className="flex-1 space-y-5 overflow-y-auto p-5">
   739	          <div className="rounded-2xl border border-violet-500/15 bg-violet-500/[0.045] p-4">
   740	            <div className="flex items-start gap-3">
   741	              <span className="mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-violet-500/10 text-violet-600 dark:text-violet-300">
   742	                <Bot className="h-4 w-4" />
   743	              </span>
   744	              <div className="space-y-1.5 text-sm leading-relaxed">
   745	                <p className="font-medium">{t("codexAssistant.greeting")}</p>
   746	                <p className="text-muted-foreground">
   747	                  {t("codexAssistant.intro")}
   748	                </p>
   749	              </div>
   750	            </div>
   751	          </div>
   752	
   753	          <section className="space-y-2">
   754	            <p className="px-1 text-xs font-medium uppercase tracking-[0.12em] text-muted-foreground">
   755	              {t("codexAssistant.readiness")}
   756	            </p>
   757	            <button
   758	              type="button"
   759	              onClick={() => !cliReady && void installCodex()}
   760	              disabled={installing || checking || cliReady}
   761	              className="flex w-full items-center gap-3 rounded-xl border border-border bg-card px-3 py-3 text-left transition hover:border-violet-500/30 disabled:cursor-default"
   762	            >
   763	              <span
   764	                className={cn(
   765	                  "flex h-8 w-8 items-center justify-center rounded-lg",
   766	                  cliReady
   767	                    ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-300"
   768	                    : "bg-amber-500/10 text-amber-600 dark:text-amber-300",
   769	                )}
   770	              >
   771	                {installing || checking ? (
   772	                  <Loader2 className="h-4 w-4 animate-spin" />
   773	                ) : cliReady ? (
   774	                  <Check className="h-4 w-4" />
   775	                ) : (
   776	                  <Terminal className="h-4 w-4" />
   777	                )}
   778	              </span>
   779	              <span className="min-w-0 flex-1">
   780	                <span className="block text-sm font-medium">
   781	                  {t("codexAssistant.codexCli")}
   782	                </span>
   783	                <span className="block text-xs text-muted-foreground">
   784	                  {cliReady
   785	                    ? t("codexAssistant.cliReady", {
   786	                        version: runtime?.version,
   787	                      })
   788	                    : t("codexAssistant.installCli")}
   789	                </span>
   790	              </span>
   791	              {!cliReady && (
   792	                <ChevronRight className="h-4 w-4 text-muted-foreground" />
   793	              )}
   794	            </button>
   795	
   796	            <button
   797	              type="button"
   798	              onClick={() => {
   799	                closeAssistant();
   800	                onOpenCodexConfiguration();
   801	              }}
   802	              className="flex w-full items-center gap-3 rounded-xl border border-border bg-card px-3 py-3 text-left transition hover:border-violet-500/30"
   803	            >
   804	              <span
   805	                className={cn(
   806	                  "flex h-8 w-8 items-center justify-center rounded-lg",
   807	                  effectiveProviderReady
   808	                    ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-300"
   809	                    : "bg-amber-500/10 text-amber-600 dark:text-amber-300",
   810	                )}
   811	              >
   812	                {effectiveProviderReady ? (
   813	                  <Check className="h-4 w-4" />
   814	                ) : (
   815	                  <CircleAlert className="h-4 w-4" />
   816	                )}
   817	              </span>
   818	              <span className="min-w-0 flex-1">
   819	                <span className="block text-sm font-medium">
   820	                  {t("codexAssistant.modelProvider")}
   821	                </span>
   822	                <span className="block text-xs text-muted-foreground">
   823	                  {effectiveProviderReady
   824	                    ? t("codexAssistant.providerReady")
   825	                    : t("codexAssistant.configureProvider")}
   826	                </span>
   827	              </span>
   828	              <ChevronRight className="h-4 w-4 text-muted-foreground" />
   829	            </button>
   830	          </section>
   831	
   832	          <section className="space-y-3">
   833	            <p className="px-1 text-xs font-medium uppercase tracking-[0.12em] text-muted-foreground">
   834	              {t("codexAssistant.chatLabel")}
   835	            </p>
   836	            {webBridgeActive && (
   837	              <p className="rounded-lg border border-sky-500/20 bg-sky-500/[0.045] px-3 py-2 text-xs leading-relaxed text-sky-800 dark:text-sky-200">
   838	                {t("codexAssistant.webBridgeNotice")}
   839	              </p>
   840	            )}
   841	
   842	            <div className="space-y-2" aria-live="polite">
   843	              {chatMessages.length === 0 ? (
   844	                <p className="rounded-lg border border-border/70 bg-muted/30 px-3 py-2 text-xs leading-relaxed text-muted-foreground">
   845	                  {t("codexAssistant.chatEmpty")}
   846	                </p>
   847	              ) : (
   848	                chatMessages.map((message, index) => (
   849	                  <div
   850	                    key={`${message.role}-${index}`}
   851	                    className={cn(
   852	                      "max-w-[92%] whitespace-pre-wrap break-words rounded-xl px-3 py-2 text-sm leading-relaxed",
   853	                      message.role === "user"
   854	                        ? "ml-auto bg-violet-500/15 text-foreground"
   855	                        : "mr-auto border border-border bg-card",
   856	                    )}
   857	                  >
   858	                    {message.content}
   859	                  </div>
   860	                ))
   861	              )}
   862	            </div>
   863	
   864	            {approvals.length > 0 && (
   865	              <div className="space-y-2" aria-live="assertive">
   866	                <p className="flex items-center gap-1.5 px-1 text-xs font-medium text-amber-700 dark:text-amber-200">
   867	                  <ShieldAlert className="h-3.5 w-3.5" />
   868	                  {t("codexAssistant.approvalRequired", {
   869	                    defaultValue: "Your approval is required",
   870	                  })}
   871	                </p>
   872	                {approvals.map((approval) => (
   873	                  <ApprovalCard
   874	                    key={approval.id}
   875	                    approval={approval}
   876	                    responding={respondingApprovalIds.has(approval.id)}
   877	                    onRespond={(decision) =>
   878	                      void respondToApproval(approval.id, decision)
   879	                    }
   880	                  />
   881	                ))}
   882	              </div>
   883	            )}
   884	
   885	            <div className="rounded-xl border border-border bg-card p-3">
   886	              <Input
   887	                value={request}
   888	                onChange={(event) => setRequest(event.target.value)}
   889	                onKeyDown={(event) => {
   890	                  if (event.key === "Enter") void sendMessage();
   891	                }}
   892	                disabled={!ready || running || startingSession}
   893	                placeholder={
   894	                  ready
   895	                    ? t("codexAssistant.chatPlaceholder")
   896	                    : (blockedReason ?? undefined)
   897	                }
   898	                className="border-0 bg-transparent px-0 shadow-none focus:ring-0"
   899	              />
   900	              <div className="mt-2 flex items-center justify-between border-t border-border/70 pt-2">
   901	                <span className="text-[11px] text-muted-foreground">
   902	                  {ready ? t("codexAssistant.chatHint") : blockedReason}
   903	                </span>
   904	                <Button
   905	                  size="icon"
   906	                  className="h-8 w-8 rounded-lg"
   907	                  disabled={
   908	                    !ready || !request.trim() || running || startingSession
   909	                  }
   910	                  onClick={() => void sendMessage()}
   911	                  aria-label={t("codexAssistant.sendChat")}
   912	                >
   913	                  {running || startingSession ? (
   914	                    <Loader2 className="h-3.5 w-3.5 animate-spin" />
   915	                  ) : (
   916	                    <Send className="h-3.5 w-3.5" />
   917	                  )}
   918	                </Button>
   919	              </div>
   920	            </div>
   921	
   922	            {running && sessionId && (
   923	              <Button
   924	                variant="outline"
   925	                size="sm"
   926	                className="w-full"
   927	                disabled={stopping}
   928	                onClick={() => void cancelRun()}
   929	              >
   930	                {stopping ? (
   931	                  <Loader2 className="h-3.5 w-3.5 animate-spin" />
   932	                ) : (
   933	                  <CircleStop className="h-3.5 w-3.5" />
   934	                )}
   935	                {stopping
   936	                  ? t("codexAssistant.cancelling")
   937	                  : t("codexAssistant.stopRun")}
   938	              </Button>
   939	            )}
   940	
   941	            {runError && (
   942	              <p className="rounded-lg border border-destructive/25 bg-destructive/5 px-3 py-2 text-xs leading-relaxed text-destructive">
   943	                {runError}
   944	              </p>
   945	            )}
   946	
   947	            {logs.length > 0 && (
   948	              <div className="overflow-hidden rounded-xl border border-border bg-muted/35">
   949	                <p className="border-b border-border px-3 py-2 text-[11px] font-medium uppercase tracking-[0.1em] text-muted-foreground">
   950	                  {t("codexAssistant.executionLog")}
   951	                </p>
   952	                <pre className="max-h-48 overflow-auto whitespace-pre-wrap break-words px-3 py-2 font-mono text-[11px] leading-relaxed text-muted-foreground">
   953	                  {logs.join("\n")}
   954	                </pre>
   955	              </div>
   956	            )}
   957	          </section>
   958	        </div>
   959	
   960	        <footer className="border-t border-border px-5 py-3 text-xs text-muted-foreground">
   961	          <span className="inline-flex items-center gap-1.5">
   962	            <MessageCircle className="h-3.5 w-3.5 text-violet-500" />
   963	            {t("codexAssistant.safetyNote")}
   964	          </span>
   965	        </footer>
   966	      </aside>
   967	    </>
   968	  );
   969	}
