   125	/// 为持有确切认证材料的调用方提供优先精确匹配、再启发式兜底的 URL 脱敏。
   126	pub(crate) fn url_for_log_with_secrets<'a>(
   127	    url: &'a str,
   128	    known_secrets: &'a [String],
   129	) -> RedactedUrl<'a> {
   130	    RedactedUrl { url, known_secrets }
   131	}

   133	/// 已知密钥参与子串脱敏的最短长度：过短的值(如 "api")当作子串会误伤无关文本，
   134	/// 所以只对足够长、几乎不可能是普通词的值做替换。
   135	const MIN_KNOWN_SECRET_LEN: usize = 8;
   136	
   137	/// 唯一的密钥脱敏原语：把字符串里出现的、我们确切握有的密钥值替换为 [REDACTED]。
   138	/// 不做任何“看起来像密钥”的形状猜测——只隐藏已知值，天然收敛、不误伤正常路径。
   139	pub(crate) fn redact_known_secrets(text: &str, known_secrets: &[String]) -> String {
   140	    redact_known_secrets_with_min_length(text, known_secrets, MIN_KNOWN_SECRET_LEN)
   141	}
   142	
   143	/// Exact-value redaction for user-visible error text. Unlike URL logging, a
   144	/// short known credential must still be hidden because the result reaches the
   145	/// UI rather than a diagnostic-only path.
   146	pub(crate) fn redact_known_secrets_strict(text: &str, known_secrets: &[String]) -> String {
   147	    redact_known_secrets_with_min_length(text, known_secrets, 1)
   148	}
   149	
   150	fn redact_known_secrets_with_min_length(
   151	    text: &str,
   152	    known_secrets: &[String],
   153	    minimum_chars: usize,
   154	) -> String {
   155	    let mut output = text.to_string();
   156	    for secret in known_secrets {
   157	        if secret.chars().count() >= minimum_chars {
   158	            output = output.replace(secret.as_str(), "[REDACTED]");
   159	        }
   160	    }
   161	    output
   162	}

   164	/// 无 scheme 的裸 authority 形态(如 `user:pass@host/path`)剥掉 userinfo：
   165	/// 仅当 `@` 出现在第一个 `/` 之前时才视为凭据。
   166	fn strip_bare_userinfo(input: &str) -> &str {
   167	    let authority_end = input.find('/').unwrap_or(input.len());
   168	    match input[..authority_end].rfind('@') {
   169	        Some(at) => &input[at + 1..],
   170	        None => input,
   171	    }
   172	}

   174	pub(crate) fn redact_url_for_log(url_str: &str) -> String {
   175	    redact_url_for_log_with_secrets(url_str, &[])
   176	}
   177	
   178	/// 为日志脱敏 URL：剥掉 userinfo(user:pass@) 与整个 query/fragment，保留
   179	/// scheme/host/port/path 供诊断(如 base_url 配错路径导致 404)，最后再抹掉已知密钥值。
   180	pub(crate) fn redact_url_for_log_with_secrets(url_str: &str, known_secrets: &[String]) -> String {
   181	    let scheme_relative = url_str.starts_with("//");
   182	    let parsed = if scheme_relative {
   183	        url::Url::parse(&format!("https:{url_str}"))
   184	    } else {
   185	        url::Url::parse(url_str)
   186	    };
   187	
   188	    let sanitized = match parsed {
   189	        Ok(mut url) if url.has_host() => {
   190	            let _ = url.set_username("");
   191	            let _ = url.set_password(None);
   192	            url.set_query(None);
   193	            url.set_fragment(None);
   194	            let rendered = url.as_str();
   195	            if scheme_relative {
   196	                rendered
   197	                    .strip_prefix("https:")
   198	                    .unwrap_or(rendered)
   199	                    .to_string()
   200	            } else {
   201	                rendered.to_string()
   202	            }
   203	        }
   204	        _ => {
   205	            // 解析失败(相对路径、含裸 userinfo 的非法 URL 等)：丢掉 query/fragment，
   206	            // 尽力剥掉 userinfo，其余原样保留。
   207	            let without_tail = url_str.split(['?', '#']).next().unwrap_or(url_str);
   208	            strip_bare_userinfo(without_tail).to_string()
   209	        }
   210	    };
   211	
   212	    redact_known_secrets(&sanitized, known_secrets)
   213	}
   214	
   215	/// 只保留 `scheme://host:port`，丢掉 path/query/userinfo。用于我们手里没有任何
   216	/// 已知密钥可脱敏 path 的场景——凭据可能整个内嵌在 base_url 的 path 里，此时
   217	/// 记录 path 无法保证不泄漏，只能退回到 origin。
   218	pub(crate) fn redact_url_origin_for_log(url_str: &str) -> String {
   219	    let scheme_relative = url_str.starts_with("//");
   220	    let parsed = if scheme_relative {
   221	        url::Url::parse(&format!("https:{url_str}"))
   222	    } else {
   223	        url::Url::parse(url_str)
   224	    };
   225	
   226	    match parsed {
   227	        Ok(url) if url.has_host() => {
   228	            let authority = &url[url::Position::BeforeHost..url::Position::AfterPort];
   229	            if scheme_relative {
   230	                format!("//{authority}")
   231	            } else {
   232	                format!("{}://{authority}", url.scheme())
   233	            }
   234	        }
   235	        _ => "[invalid target]".to_string(),
   236	    }
   237	}

   243	/// 统一处理 ccswitch:// 深链接 URL
   244	///
   245	/// - 解析 URL
   246	/// - 向前端发射 `deeplink-import` / `deeplink-error` 事件
   247	/// - 可选：在成功时聚焦主窗口
   248	fn handle_deeplink_url(
   249	    app: &tauri::AppHandle,
   250	    url_str: &str,
   251	    focus_main_window: bool,
   252	    source: &str,
   253	) -> bool {
   254	    if !url_str.starts_with("ccswitch://") {
   255	        return false;
   256	    }
   257	
   258	    log::info!(
   259	        "✓ Deep link URL detected from {source}: {}",
   260	        url_for_log(url_str)
   261	    );
   262	
   263	    match crate::deeplink::parse_deeplink_url(url_str) {
   264	        Ok(request) => {
   265	            log::info!(
   266	                "✓ Successfully parsed deep link: resource={}, app={:?}, name={:?}",
   267	                request.resource,
   268	                request.app,
   269	                request.name
   270	            );
   271	
   272	            if let Err(e) = app.emit("deeplink-import", &request) {
   273	                log::error!("✗ Failed to emit deeplink-import event: {e}");
   274	            } else {
   275	                log::info!("✓ Emitted deeplink-import event to frontend");
   276	            }
   277	
   278	            if focus_main_window {
   279	                if let Some(window) = app.get_webview_window("main") {
   280	                    let _ = window.unminimize();
   281	                    let _ = window.show();
   282	                    let _ = window.set_focus();
   283	                    #[cfg(target_os = "linux")]
   284	                    {
   285	                        linux_fix::nudge_main_window(window.clone());
   286	                    }
   287	                    log::info!("✓ Window shown and focused");
   288	                }
   289	            }
   290	        }
   291	        Err(e) => {
   292	            log::error!("✗ Failed to parse deep link URL: {e}");
   293	
   294	            if let Err(emit_err) = app.emit(
   295	                "deeplink-error",
   296	                serde_json::json!({
   297	                    "url": url_str,
   298	                    "error": e.to_string()
   299	                }),
   300	            ) {
   301	                log::error!("✗ Failed to emit deeplink-error event: {emit_err}");
   302	            }
   303	        }
   304	    }
   305	
   306	    true
   307	}

   311	async fn update_tray_menu(
   312	    app: tauri::AppHandle,
   313	    state: tauri::State<'_, AppState>,
   314	) -> Result<bool, String> {
   315	    match tray::create_tray_menu(&app, state.inner()) {
   316	        Ok(new_menu) => {
   317	            if let Some(tray) = app.tray_by_id(tray::TRAY_ID) {

   319	                    .map_err(|e| format!("更新托盘菜单失败: {e}"))?;

   321	            }
   322	            Ok(false)
   323	        }
   324	        Err(err) => {
   325	            log::error!("创建托盘菜单失败: {err}");
   326	            Ok(false)
   327	        }
   328	    }
   329	}

   344	#[cfg_attr(mobile, tauri::mobile_entry_point)]
   345	pub fn run() {
   346	    // 设置 panic hook，在应用崩溃时记录日志到 <app_config_dir>/crash.log（默认 ~/.cc-switch/crash.log）
   347	    panic_hook::setup_panic_hook();
   348	
   349	    let mut builder = tauri::Builder::default();
   350	
   351	    #[cfg(any(target_os = "macos", target_os = "windows", target_os = "linux"))]
   352	    {
   353	        builder = builder.plugin(tauri_plugin_single_instance::init(|app, args, _cwd| {
   354	            log::info!("=== Single Instance Callback Triggered ===");
   355	            log::debug!("Args count: {}", args.len());
   356	            for (i, arg) in args.iter().enumerate() {
   357	                log::debug!("  arg[{i}]: {}", url_for_log(arg));
   358	            }
   359	
   360	            if crate::lightweight::is_lightweight_mode() {
   361	                if let Err(e) = crate::lightweight::exit_lightweight_mode(app) {
   362	                    log::error!("退出轻量模式重建窗口失败: {e}");
   363	                }
   364	            }
   365	
   366	            // Check for deep link URL in args (mainly for Windows/Linux command line)
   367	            let mut found_deeplink = false;
   368	            for arg in &args {
   369	                if handle_deeplink_url(app, arg, false, "single_instance args") {
   370	                    found_deeplink = true;
   371	                    break;
   372	                }
   373	            }
   374	
   375	            if !found_deeplink {
   376	                log::info!("ℹ No deep link URL found in args (this is expected on macOS when launched via system)");
   377	            }
   378	
   379	            // Show and focus window regardless
   380	            if let Some(window) = app.get_webview_window("main") {
   381	                let _ = window.unminimize();
   382	                let _ = window.show();
   383	                let _ = window.set_focus();
   384	                #[cfg(target_os = "linux")]
   385	                {
   386	                    linux_fix::nudge_main_window(window.clone());
   387	                }
   388	            }
   389	        }));
   390	    }
   391	
   392	    #[cfg(target_os = "windows")]
   393	    {
   394	        let startup_page_handled = AtomicBool::new(false);
   395	        builder = builder.on_page_load(move |webview, payload| {
   396	            if webview.label() == "main"
   397	                && payload.event() == tauri::webview::PageLoadEvent::Finished
   398	                && payload.url().scheme() != "about"
   399	                && !startup_page_handled.swap(true, Ordering::Relaxed)
   400	                && !crate::settings::get_settings().silent_startup
   401	            {
   402	                let _ = webview.window().show();
   403	                log::info!("主页面加载完成，主窗口已显示");
   404	            }
   405	        });
   406	    }
   407	
   408	    let builder = builder
   409	        // 注册 deep-link 插件（处理 macOS AppleEvent 和其他平台的深链接）
   410	        .plugin(tauri_plugin_deep_link::init())
   411	        // 拦截窗口关闭：根据设置决定是否最小化到托盘
   412	        .on_window_event(|window, event| {
   413	            if let tauri::WindowEvent::CloseRequested { api, .. } = event {
   414	                // 数据库版本过新的恢复模式下没有托盘可唤回，关闭即退出，避免应用隐身后台
   415	                let in_db_recovery = crate::init_status::get_init_error()
   416	                    .map(|p| p.kind.as_deref() == Some("db_version_too_new"))
   417	                    .unwrap_or(false);
   418	                if in_db_recovery {
   419	                    api.prevent_close();
   420	                    window.app_handle().exit(0);
   421	                    return;
   422	                }
   423	
   424	                let settings = crate::settings::get_settings();
   425	
   426	                if settings.minimize_to_tray_on_close {
   427	                    api.prevent_close();
   428	                    let _ = window.hide();
   429	                    #[cfg(target_os = "windows")]
   430	                    {
   431	                        let _ = window.set_skip_taskbar(true);
   432	                    }
   433	                    #[cfg(target_os = "macos")]
   434	                    {
   435	                        tray::apply_tray_policy(window.app_handle(), false);
   436	                    }
   437	                } else {
   438	                    api.prevent_close();
   439	                    window.app_handle().exit(0);
   440	                }
   441	            }
   442	        })
   443	        .plugin(tauri_plugin_process::init())
   444	        .plugin(tauri_plugin_dialog::init())
   445	        .plugin(tauri_plugin_opener::init())
   446	        .plugin(tauri_plugin_store::Builder::new().build())
   447	        .plugin(
   448	            tauri_plugin_window_state::Builder::default()
   449	                .with_state_flags(window_state_flags())
   450	                .build(),
   451	        )
   452	        .setup(|app| {
   453	            let _ = rustls::crypto::ring::default_provider().install_default();
   454	
   455	            // 预先刷新 Store 覆盖配置，确保后续路径读取正确（日志/数据库等）
   456	            app_store::refresh_app_config_dir_override(app.handle());
   457	            panic_hook::init_app_config_dir(crate::config::get_app_config_dir());
   458	
   459	            // 初始化日志（输出到 <app_config_dir>/logs/cc-switch.log）
   460	            {
   461	                use tauri_plugin_log::{RotationStrategy, Target, TargetKind, TimezoneStrategy};
   462	
   463	                let log_dir = panic_hook::get_log_dir();
   464	
   465	                // 确保日志目录存在
   466	                if let Err(e) = std::fs::create_dir_all(&log_dir) {
   467	                    eprintln!("创建日志目录失败: {e}");
   468	                }
   469	
   470	                app.handle().plugin(
   471	                    tauri_plugin_log::Builder::default()
   472	                        // 底层保留 Trace 能力，便于加载用户配置后动态调高级别。
   473	                        // 插件注册后会立即把全局级别收紧到 Info，避免启动阶段全量 Trace。
   474	                        .level(log::LevelFilter::Trace)
   475	                        // plugin-log 的前端 command 会直达 logger，绕过 log 宏的全局
   476	                        // max_level；在分发层补一次过滤，确保动态总开关同样约束前端日志。
   477	                        .filter(|metadata| {
   478	                            runtime_log_level_allows(metadata.level(), log::max_level())
   479	                        })
   480	                        .targets([
   481	                            Target::new(TargetKind::Stdout),
   482	                            Target::new(TargetKind::Folder {
   483	                                path: log_dir,
   484	                                file_name: Some("cc-switch".into()),
   485	                            }),
   486	                        ])
   487	                        // KeepSome(4) 保留 4 个轮转归档，加上当前文件最多约 100 MiB。
   488	                        // 轮转仅按大小触发；跨重启继续追加，不再丢失上一次运行的日志。
   489	                        .rotation_strategy(RotationStrategy::KeepSome(4))
   490	                        .max_file_size(20 * 1024 * 1024)
   491	                        .timezone_strategy(TimezoneStrategy::UseLocal)
   492	                        .build(),
   493	                )?;
   494	
   495	                // 用户配置存在数据库中，数据库尚未打开时使用保守的 Info 级别。
   496	                log::set_max_level(log::LevelFilter::Info);
   497	                log::info!("=== CC Switch v{} started ===", env!("CARGO_PKG_VERSION"));
   498	            }
   499	
   500	            // 首次读取覆盖路径时 logger 尚未可用；此处重放一次，
   501	            // 让 Store 损坏或路径无效等启动警告能够真正落盘。

  1852	                // 互等造成进程永久卡死（更新已安装但应用冻结、不再重启，见 #3998）。
  1853	                //
  1854	                // 重启路径交还 Tauri 默认流程即可：
  1855	                //   - 窗口状态：插件 Exit 钩子在主线程保存（同线程读取窗口几何，无死锁）
  1856	                //   - 托盘图标：Tauri 内部 cleanup_before_exit 清理，正常走 Drop
  1857	                //   - 代理/Live 配置：无需恢复，重启后新实例立即接管并恢复代理状态
  1858	                //   - 100ms 落盘等待：重启前的 DB 写入均为命令驱动、此刻已完成，
  1859	                //     与所有 Tauri 应用默认重启路径的行为一致，无需额外等待
  1860	                ExitRequestAction::DeferToTauriRestart => {
  1861	                    log::info!("收到重启请求 (code={code:?})，交由 Tauri 默认重启流程 re-exec");
  1862	                    return;
  1863	                }
  1864	                // 其它 Some(_)：用户主动调用 app.exit() 退出（如托盘菜单"退出"），
  1865	                // 此时执行清理后退出。
  1866	                ExitRequestAction::CleanupAndExit => {}
  1867	            }
  1868	
  1869	            log::info!("收到用户主动退出请求 (code={code:?})，开始清理...");
  1870	            api.prevent_exit();
  1871	
  1872	            let app_handle = app_handle.clone();
  1873	            tauri::async_runtime::spawn(async move {
  1874	                save_window_state_before_exit(&app_handle);
  1875	                cleanup_before_exit(&app_handle).await;
  1876	                // 先于 std::process::exit 显式移除托盘图标。
  1877	                // 进程直接退出时 Tauri 运行时不走正常 Drop 流程，
  1878	                // 不会向 Windows Shell 发送 NIM_DELETE，导致已退出的进程
  1879	                // 注册的图标仍残留在系统托盘（鼠标悬停 Shell 才会重绘发现进程已死）。
  1880	                remove_tray_icon_before_exit(&app_handle);
  1881	                log::info!("清理完成，退出应用");
  1882	
  1883	                // 短暂等待确保所有 I/O 操作（如数据库写入）刷新到磁盘
  1884	                tokio::time::sleep(std::time::Duration::from_millis(100)).await;
  1885	
  1886	                // 使用 std::process::exit 避免再次触发 ExitRequested
  1887	                std::process::exit(0);
  1888	            });
  1889	            return;
  1890	        }
  1891	
  1892	        #[cfg(target_os = "macos")]
  1893	        {
  1894	            match event {
  1895	                // macOS 在 Dock 图标被点击并重新激活应用时会触发 Reopen 事件，这里手动恢复主窗口
  1896	                RunEvent::Reopen { .. } => {
  1897	                    if let Some(window) = app_handle.get_webview_window("main") {
  1898	                        #[cfg(target_os = "windows")]
  1899	                        {
  1900	                            let _ = window.set_skip_taskbar(false);
  1901	                        }
  1902	                        let _ = window.unminimize();
  1903	                        let _ = window.show();
  1904	                        let _ = window.set_focus();
  1905	                        tray::apply_tray_policy(app_handle, true);
  1906	                    } else if crate::lightweight::is_lightweight_mode() {
  1907	                        if let Err(e) = crate::lightweight::exit_lightweight_mode(app_handle) {
  1908	                            log::error!("退出轻量模式重建窗口失败: {e}");
  1909	                        }
  1910	                    }
  1911	                }
  1912	                // 处理通过自定义 URL 协议触发的打开事件（例如 ccswitch://...）
  1913	                RunEvent::Opened { urls } => {
  1914	                    if let Some(url) = urls.first() {
  1915	                        let url_str = url.to_string();
  1916	                        log::info!(
  1917	                            "RunEvent::Opened with URL: {}",
  1918	                            url_for_log(&url_str)
  1919	                        );
  1920	
  1921	                        if url_str.starts_with("ccswitch://") {
  1922	                            if crate::lightweight::is_lightweight_mode() {
  1923	                                if let Err(e) = crate::lightweight::exit_lightweight_mode(app_handle)
  1924	                                {
  1925	                                    log::error!("退出轻量模式重建窗口失败: {e}");
  1926	                                }
  1927	                            }
  1928	
  1929	                            // 解析并广播深链接事件，复用与 single_instance 相同的逻辑
  1930	                            match crate::deeplink::parse_deeplink_url(&url_str) {
  1931	                                Ok(request) => {
  1932	                                    log::info!(
  1933	                                        "Successfully parsed deep link from RunEvent::Opened: resource={}, app={:?}",
  1934	                                        request.resource,
  1935	                                        request.app
  1936	                                    );
  1937	
  1938	                                    if let Err(e) =
  1939	                                        app_handle.emit("deeplink-import", &request)
  1940	                                    {
  1941	                                        log::error!(
  1942	                                            "Failed to emit deep link event from RunEvent::Opened: {e}"
  1943	                                        );
  1944	                                    }
  1945	                                }
  1946	                                Err(e) => {
  1947	                                    log::error!(
  1948	                                        "Failed to parse deep link URL from RunEvent::Opened: {e}"
  1949	                                    );
  1950	
  1951	                                    if let Err(emit_err) = app_handle.emit(
  1952	                                        "deeplink-error",
  1953	                                        serde_json::json!({
  1954	                                            "url": url_str,
  1955	                                            "error": e.to_string()
  1956	                                        }),
  1957	                                    ) {
  1958	                                        log::error!(
  1959	                                            "Failed to emit deep link error event from RunEvent::Opened: {emit_err}"
  1960	                                        );
  1961	                                    }
  1962	                                }
  1963	                            }
  1964	
  1965	                            // 确保主窗口可见
  1966	                            if let Some(window) = app_handle.get_webview_window("main") {
  1967	                                let _ = window.unminimize();
  1968	                                let _ = window.show();
  1969	                                let _ = window.set_focus();
  1970	                            }
  1971	                        }
  1972	                    }
  1973	                }
  1974	                _ => {}
  1975	            }
  1976	        }
  1977	
  1978	        #[cfg(not(target_os = "macos"))]
  1979	        {
  1980	            let _ = (app_handle, event);
  1981	        }
  1982	    });
  1983	}

  1989	/// 应用退出前的清理工作
  1990	///
  1991	/// 在应用退出前检查代理服务器状态，如果正在运行则停止代理并恢复 Live 配置。
  1992	/// 确保 Claude Code/Codex/Gemini 的配置不会处于损坏状态。
  1993	/// 使用 stop_with_restore_keep_state 保留 settings 表中的代理状态，下次启动时自动恢复。
  1994	pub async fn cleanup_before_exit(app_handle: &tauri::AppHandle) {
  1995	    commands::shutdown_codex_assistant_sessions();
  1996	    if let Some(state) = app_handle.try_state::<store::AppState>() {
  1997	        let proxy_service = &state.proxy_service;
  1998	
  1999	        if crate::config::is_test_sandbox() {
  2000	            if proxy_service.is_running().await {
  2001	                if let Err(e) = proxy_service.stop().await {
  2002	                    log::error!("开发沙箱退出时停止本地代理失败: {e}");
  2003	                }
  2004	            }
  2005	            return;
  2006	        }
  2007	
  2008	        // 退出时也需要兜底：代理可能已崩溃/未运行，但 Live 接管残留仍在（占位符/备份）。
  2009	        let has_backups = match state.db.has_any_live_backup().await {
  2010	            Ok(v) => v,
  2011	            Err(e) => {
  2012	                log::error!("退出时检查 Live 备份失败: {e}");
  2013	                false
  2014	            }
  2015	        };
  2016	        let live_taken_over = proxy_service.detect_takeover_in_live_configs();
  2017	        let needs_restore = has_backups || live_taken_over;
  2018	
  2019	        if needs_restore {
  2020	            log::info!("检测到接管残留，开始恢复 Live 配置（保留代理状态）...");
  2021	            // 使用 keep_state 版本，保留 settings 表中的代理状态
  2022	            if let Err(e) = proxy_service.stop_with_restore_keep_state().await {
  2023	                log::error!("退出时恢复 Live 配置失败: {e}");
  2024	            } else {
  2025	                log::info!("已恢复 Live 配置（代理状态已保留，下次启动将自动恢复）");
  2026	            }
  2027	            return;
  2028	        }
  2029	
  2030	        // 非接管模式：代理在运行则仅停止代理
  2031	        if proxy_service.is_running().await {
  2032	            log::info!("检测到代理服务器正在运行，开始停止...");
  2033	            if let Err(e) = proxy_service.stop().await {
  2034	                log::error!("退出时停止代理失败: {e}");
  2035	            }
  2036	            log::info!("代理服务器清理完成");
  2037	        }
  2038	    }
  2039	}

  2041	/// 主动从系统托盘移除托盘图标。
  2042	///
  2043	/// `std::process::exit` 会绕过 Tauri 运行时，触发不了 `TrayIcon::drop()`，

  2046	///
  2047	/// 通过 `set_visible(false)` 走 `WM_USER_HIDE_TRAYICON` 消息路径，
  2048	/// 触发 tray-icon 内部的 `remove_tray_icon` → `Shell_NotifyIconW(NIM_DELETE)`，
  2049	/// 在进程结束前干净地把图标摘掉。其它平台 `set_visible(false)` 也是
  2050	/// 正常的隐藏/移除语义，作为跨平台兜底也安全。
  2051	pub(crate) fn remove_tray_icon_before_exit(app_handle: &tauri::AppHandle) {
  2052	    if let Some(tray) = app_handle.tray_by_id(tray::TRAY_ID) {
  2053	        if let Err(e) = tray.set_visible(false) {
  2054	            log::warn!("退出时移除托盘图标失败: {e}");
  2055	        } else {
  2056	            log::info!("已显式从系统托盘移除图标");
  2057	        }
  2058	    }
  2059	}

  2115	                    .proxy_service
  2116	                    .set_takeover_for_app(app_type, false)

  2335	// 退出请求分类
  2336	// ============================================================
  2337	
  2338	/// `RunEvent::ExitRequested` 的三类来源，处理方式必须区分。
  2339	///
  2340	/// 关键约束：重启请求（`code == RESTART_EXIT_CODE`）上 `prevent_exit()` 会被
  2341	/// Tauri 静默忽略（见 `ExitRequestApi::prevent_exit` 文档），事件循环必定继续
  2342	/// 退出并触发各插件的 `RunEvent::Exit` 钩子；任何与之并发的自定义清理任务都
  2343	/// 可能与插件退出钩子争用同一状态而死锁。
  2344	#[derive(Debug, Clone, Copy, PartialEq, Eq)]
  2345	enum ExitRequestAction {
  2346	    /// `code` 为 `None`：运行时自动触发（如隐藏窗口的 WebView 被回收导致无存活
  2347	    /// 窗口），阻止退出、保持托盘后台运行。
  2348	    StayInTray,
  2349	    /// `code` 为 `RESTART_EXIT_CODE`：`app.restart()` / 自更新 relaunch 发起的
  2350	    /// 重启，不拦截、不做自定义清理，交还 Tauri 默认 re-exec 流程。
  2351	    DeferToTauriRestart,
  2352	    /// 其它 `Some(_)`：用户主动退出（托盘「退出」等），执行完整异步清理后结束进程。
  2353	    CleanupAndExit,
  2354	}
  2355	
  2356	fn classify_exit_request(code: Option<i32>) -> ExitRequestAction {
  2357	    match code {
  2358	        None => ExitRequestAction::StayInTray,
  2359	        Some(tauri::RESTART_EXIT_CODE) => ExitRequestAction::DeferToTauriRestart,
  2360	        Some(_) => ExitRequestAction::CleanupAndExit,
  2361	    }
  2362	}

  2365	// 在应用主动退出前显式持久化窗口状态
  2366	// ============================================================
  2367	
  2368	fn window_state_flags() -> StateFlags {
  2369	    StateFlags::POSITION | StateFlags::SIZE | StateFlags::MAXIMIZED
  2370	}
  2371	
  2372	/// 当前应用的退出路径会拦截 `ExitRequested` 并最终直接 `std::process::exit(0)`，
  2373	/// 这里需要在真正结束进程前手动落盘，避免 window-state 插件的默认退出钩子被绕过。
  2374	pub fn save_window_state_before_exit(app_handle: &tauri::AppHandle) {
  2375	    if let Err(err) = app_handle.save_window_state(window_state_flags()) {
  2376	        log::error!("退出前保存窗口状态失败: {err}");
  2377	    } else {
  2378	        log::info!("已在退出前保存窗口状态");
  2379	    }
  2380	}

  2382	/// 主动释放 single-instance 锁。
  2383	///
  2384	/// macOS single-instance 使用 `/tmp/{identifier}.sock`。我们有若干路径会直接
  2385	/// `std::process::exit(0)`，不会触发插件挂在 `RunEvent::Exit` 上的清理钩子。
  2386	/// 重启前主动 destroy 可以避免新进程误连旧 listener 后自行退出。
  2387	pub fn destroy_single_instance_lock(app_handle: &tauri::AppHandle) {
  2388	    #[cfg(any(target_os = "macos", target_os = "windows", target_os = "linux"))]
  2389	    tauri_plugin_single_instance::destroy(app_handle);
  2390	}
  2391	
  2392	/// 清理托盘图标、释放 single-instance 锁后重启当前应用。
  2393	///
  2394	/// 直接走 `tauri::process::restart`（spawn 新进程 + `exit(0)`），不经过事件
  2395	/// 循环退出，因此 Tauri 内部的 `cleanup_before_exit` 和各插件的
  2396	/// `RunEvent::Exit` 钩子都不会执行。需要的清理由调用方与本函数显式补偿：
  2397	/// 窗口状态、代理/Live 恢复（调用方）；托盘图标、single-instance 锁（本函数）。
  2398	///
  2399	/// 有意不调 `AppHandle::cleanup_before_exit()`：它会在调用线程上 Drop 托盘
  2400	/// 图标，而 macOS 的 NSStatusItem 操作要求主线程；`set_visible(false)` 走
  2401	/// `run_item_main_thread` 代理，跨线程安全（见 `remove_tray_icon_before_exit`）。
  2402	pub fn restart_process(app_handle: &tauri::AppHandle) -> ! {
  2403	    commands::shutdown_codex_assistant_sessions();
  2404	    remove_tray_icon_before_exit(app_handle);
  2405	    destroy_single_instance_lock(app_handle);
  2406	    tauri::process::restart(&app_handle.env());
  2407	}
