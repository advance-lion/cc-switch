   117	  const commandOutput = async (
   118	    executable: string,
   119	    args: string[],
   120	    label: string,
   121	  ): Promise<string> =>
   122	    await new Promise((resolve, reject) => {
   123	      const child = spawn(executable, args, {
   124	        stdio: ["ignore", "pipe", "pipe"],
   125	        windowsHide: true,
   126	      });
   127	      let stdout = "";
   128	      let stderr = "";
   129	      let settled = false;
   130	      child.stdout.on("data", (chunk) => (stdout += chunk));
   131	      child.stderr.on("data", (chunk) => (stderr += chunk));
   132	      child.on("error", (error) => {
   133	        if (settled) return;
   134	        settled = true;
   135	        reject(new Error(`Failed to start ${label}: ${error.message}`));
   136	      });
   137	      child.on("close", (code) => {
   138	        if (settled) return;
   139	        settled = true;
   140	        const output = stdout.trim();
   141	        const errorOutput = stderr.trim();
   142	        if (code !== 0) {
   143	          reject(
   144	            new Error(
   145	              `${label} failed with exit code ${code ?? -1}${
   146	                errorOutput || output ? `: ${errorOutput || output}` : ""
   147	              }`,
   148	            ),
   149	          );
   150	          return;
   151	        }
   152	        resolve(output);
   153	      });
   154	    });

   324	  const MAX_CHAT_HISTORY_TURNS = 12;
   325	  const MAX_CHAT_TURN_LENGTH = 2_000;
   326	  const MAX_CHAT_HISTORY_CHARS = 12_000;
   327	
   328	  type ChatTurn = { role: "user" | "assistant"; content: string };

   330	  // Mirrors normalize_chat_history in src-tauri/src/commands/codex_assistant.rs.
   331	  const normalizeChatHistory = (history: unknown): ChatTurn[] => {
   332	    if (history === undefined || history === null) return [];
   333	    if (!Array.isArray(history)) throw new Error("对话历史格式不正确");
   334	    let normalized: ChatTurn[] = [];
   335	    for (const entry of history) {
   336	      if (!entry || typeof entry !== "object")
   337	        throw new Error("对话历史格式不正确");
   338	      const { role, content } = entry as { role?: unknown; content?: unknown };
   339	      if (role !== "user" && role !== "assistant")
   340	        throw new Error("对话历史包含非法角色");
   341	      if (typeof content !== "string") throw new Error("对话历史格式不正确");
   342	      const trimmed = content.trim();
   343	      if (!trimmed) continue;
   344	      normalized.push({
   345	        role,
   346	        content: Array.from(trimmed).slice(0, MAX_CHAT_TURN_LENGTH).join(""),
   347	      });
   348	    }
   349	    if (normalized.length > MAX_CHAT_HISTORY_TURNS) {
   350	      normalized = normalized.slice(-MAX_CHAT_HISTORY_TURNS);
   351	    }
   352	    while (
   353	      normalized.reduce((total, turn) => total + turn.content.length, 0) >
   354	      MAX_CHAT_HISTORY_CHARS
   355	    ) {
   356	      normalized.shift();
   357	    }
   358	    return normalized;
   359	  };

   472	    configureServer(server: {
   473	      middlewares: {
   474	        use: (

   483	      server.middlewares.use((request, response, next) => {
   484	        if (!request.url?.startsWith(WEB_ASSISTANT_BASE)) return next();
   485	        const requestUrl = new URL(request.url, "http://127.0.0.1");
   486	        const route = requestUrl.pathname;
   487	        if (
   488	          request.method === "GET" &&

   498	          request.on("close", () => listeners.delete(response));

   503	          route === `${WEB_ASSISTANT_BASE}/recent-events`
   504	        ) {
   505	          return sendJson(response, 200, { events: recentEvents });
   506	        }
   507	        if (
   508	          request.method === "GET" &&
   509	          route === `${WEB_ASSISTANT_BASE}/status`
   510	        ) {
   511	          void (async () => {
   512	            const result = await probeToolVersion("codex");

   518	          return;
   519	        }

   522	          route === `${WEB_ASSISTANT_BASE}/desktop-status`

   524	          void detectDesktopApp(DESKTOP_APPS["codex-desktop"])
   525	            .then((status) =>
   526	              sendJson(response, 200, {
   527	                installed: status.installed,
   528	                version: status.version,
   529	                path: status.path,
   530	              }),
   531	            )
   532	            .catch((error) =>
   533	              sendJson(response, 400, {
   534	                error: error instanceof Error ? error.message : String(error),
   535	              }),
   536	            );
   537	          return;
   538	        }
   539	        if (
   540	          request.method === "GET" &&
   541	          route === `${WEB_ASSISTANT_BASE}/desktop-app-status`
   542	        ) {
   543	          void Promise.resolve()
   544	            .then(() =>
   545	              detectDesktopApp(
   546	                desktopManifest(requestUrl.searchParams.get("app")),
   547	              ),
   548	            )
   549	            .then((status) => sendJson(response, 200, status))
   550	            .catch((error) =>
   551	              sendJson(response, 400, {
   552	                error: error instanceof Error ? error.message : String(error),
   553	              }),
   554	            );
   555	          return;
   556	        }
   557	        if (request.method !== "POST")
   558	          return sendJson(response, 405, { error: "Method not allowed" });
   559	        void readJson(request)
   560	          .then(async (body) => {
   561	            if (
   562	              route === `${WEB_ASSISTANT_BASE}/session` ||
   563	              route === `${WEB_ASSISTANT_BASE}/sessions`
   564	            ) {
   565	              const session = createSession(body.history);
   566	              emit({ sessionId: session.id, kind: "session", status: "ready" });
   567	              return sendJson(response, 201, { sessionId: session.id });
   568	            }
   569	            if (
   570	              route === `${WEB_ASSISTANT_BASE}/chat` ||
   571	              route === `${WEB_ASSISTANT_BASE}/message` ||
   572	              route === `${WEB_ASSISTANT_BASE}/messages`
   573	            ) {
   574	              const requestText =
   575	                typeof body.request === "string"
   576	                  ? body.request
   577	                  : typeof body.message === "string"
   578	                    ? body.message
   579	                    : typeof body.content === "string"
   580	                      ? body.content
   581	                      : "";
   582	              if (!requestText.trim()) throw new Error("请输入想问的内容");
   583	              if (requestText.length > MAX_REQUEST_LENGTH)
   584	                throw new Error("请求过长");
   585	
   586	              const session = resolveSession(body);
   587	              const requestContent = requestText.trim();
   588	              const suppliedHistory =
   589	                body.history === undefined
   590	                  ? session.history
   591	                  : normalizeChatHistory(body.history);
   592	              const history = normalizeChatHistory(suppliedHistory);
   593	
   594	              if (looksLikeOperationRequest(requestContent)) {
   595	                const previewMessage =
   596	                  "这是浏览器预览。我可以展示操作确认，但不会安装软件、修改配置或执行真实命令。";
   597	                session.history = normalizeChatHistory([
   598	                  ...history,
   599	                  { role: "user", content: requestContent },
   600	                  { role: "assistant", content: previewMessage },
   601	                ]);
   602	                emitPreviewApproval(session, requestContent);
   603	                return sendJson(response, 202, {
   604	                  accepted: true,
   605	                  sessionId: session.id,
   606	                });
   607	              }
   608	
   609	              const answer =
   610	                "这是浏览器开发预览中的模拟回答。我可以在同一个对话里帮助你了解 AI Agent、安装方式、Provider 和配置流程；涉及操作时会先展示确认卡，而且预览不会执行真实命令或修改系统。";
   611	              session.history = normalizeChatHistory([
   612	                ...history,
   613	                { role: "user", content: requestContent },
   614	                { role: "assistant", content: answer },
   615	              ]);
   616	              emit({ sessionId: session.id, kind: "started" });
   617	              emit({
   618	                sessionId: session.id,
   619	                kind: "message",
   620	                message: answer,
   621	              });
   622	              emit({
   623	                sessionId: session.id,
   624	                kind: "finished",
   625	                success: true,
   626	                cancelled: false,
   627	                message: answer,
   628	              });
   629	              return sendJson(response, 202, {
   630	                accepted: true,
   631	                sessionId: session.id,
   632	              });
   633	            }
   634	            if (
   635	              route === `${WEB_ASSISTANT_BASE}/approval` ||
   636	              route === `${WEB_ASSISTANT_BASE}/approvals`
   637	            ) {
   638	              const session = getSession(body.sessionId ?? body.session_id);
   639	              if (!session) throw new Error("Codex 对话不存在或已失效");
   640	              const approvalId = body.approvalId ?? body.approval_id;
   641	              if (
   642	                typeof approvalId !== "string" ||
   643	                approvalId !== session.pendingApprovalId
   644	              ) {
   645	                throw new Error("确认请求不存在或已失效");
   646	              }
   647	              const decision = approvalDecision(body);
   648	              delete session.pendingApprovalId;
   649	              const message =
   650	                decision === "accept" || decision === "acceptForSession"
   651	                  ? "已记录允许。浏览器预览不会执行真实操作。"
   652	                  : decision === "cancel"
   653	                    ? "已取消任务。浏览器预览未执行任何更改。"
   654	                    : "已拒绝操作。浏览器预览未执行任何更改。";
   655	              session.history = normalizeChatHistory([
   656	                ...session.history,
   657	                { role: "assistant", content: message },
   658	              ]);
   659	              emit({
   660	                sessionId: session.id,
   661	                kind: "message",
   662	                message,
   663	              });
   664	              emit({
   665	                sessionId: session.id,
   666	                kind: "finished",
   667	                success: true,
   668	                cancelled: decision === "cancel",
   669	                message,
   670	              });
   671	              return sendJson(response, 200, {
   672	                accepted: true,
   673	                sessionId: session.id,
   674	                approvalId,
   675	                decision,
   676	                simulated: true,
   677	                message,
   678	              });
   679	            }
   680	            if (route === `${WEB_ASSISTANT_BASE}/cancel`) {
   681	              const session = getSession(
   682	                body.sessionId ?? body.session_id ?? body.runId,
   683	              );
   684	              if (!session || !session.pendingApprovalId) {
   685	                throw new Error("没有正在运行的 Codex 任务");
   686	              }
   687	              delete session.pendingApprovalId;
   688	              emit({
   689	                sessionId: session.id,
   690	                kind: "finished",
   691	                success: false,
   692	                cancelled: true,
   693	                message: "已取消任务。浏览器预览未执行任何更改。",
   694	              });
   695	              return sendJson(response, 200, { cancelled: true });
   696	            }
   697	            if (route === `${WEB_ASSISTANT_BASE}/close`) {
   698	              const sessionId = body.sessionId ?? body.session_id;
   699	              if (typeof sessionId !== "string")
   700	                throw new Error("Codex 对话不存在或已失效");
   701	              const closed = sessions.delete(sessionId);
   702	              return sendJson(response, 200, { closed });
   703	            }
   704	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-check-updates`) {
   705	              const status = await checkDesktopAppUpdates(
   706	                desktopManifest(body.app),
   707	              );
   708	              return sendJson(response, 200, status);
   709	            }
   710	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-action`) {
   711	              const manifest = desktopManifest(body.app);
   712	              if (
   713	                body.action !== "install" &&
   714	                body.action !== "update" &&
   715	                body.action !== "uninstall"
   716	              ) {
   717	                throw new Error(
   718	                  `Unsupported desktop lifecycle action: ${String(body.action ?? "")}`,
   719	                );
   720	              }
   721	              const status = await detectDesktopApp(manifest);
   722	              return sendJson(response, 200, {
   723	                ...status,
   724	                reason: `${manifest.displayName} ${body.action} is not executed in the browser preview`,
   725	              });
   726	            }
   727	            if (route === `${WEB_ASSISTANT_BASE}/desktop-app-launch`) {
   728	              await launchDesktopApp(desktopManifest(body.app));
   729	              return sendJson(response, 200, { launched: true });
   730	            }
   731	            if (route === `${WEB_ASSISTANT_BASE}/launch-desktop`) {
   732	              await launchDesktopApp(DESKTOP_APPS["codex-desktop"]);
   733	              return sendJson(response, 200, { launched: true });
   734	            }
   735	            sendJson(response, 404, { error: "Unknown bridge endpoint" });
   736	          })
   737	          .catch((error) =>
   738	            sendJson(response, 400, {
   739	              error: error instanceof Error ? error.message : String(error),
   740	            }),
   741	          );
   742	      });
   743	    },
   744	  };
   745	}
