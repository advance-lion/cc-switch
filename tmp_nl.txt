   151	fn validate_input(input: &str) -> Result<String, String> {
   152	    let input = input.trim();
   153	    if input.is_empty() {
   154	        return Err("请输入消息".to_string());
   155	    }
   156	    if input.chars().count() > MAX_INPUT_LENGTH {
   157	        return Err(format!("消息过长（最多 {MAX_INPUT_LENGTH} 个字符）"));
   158	    }
   159	    Ok(input.to_string())
   160	}

   162	fn resolve_workspace_dir(raw: Option<&str>) -> Result<PathBuf, String> {
   163	    let path = match raw.map(str::trim).filter(|value| !value.is_empty()) {
   164	        Some(raw) => fs::canonicalize(raw).map_err(|error| format!("无法访问工作目录: {error}"))?,
   165	        None => {
   166	            let path = crate::config::get_app_config_dir()
   167	                .join("codex-assistant")
   168	                .join("workspace");
   169	            fs::create_dir_all(&path).map_err(|error| format!("无法创建助手工作目录: {error}"))?;
   170	            fs::canonicalize(path).map_err(|error| format!("无法访问助手工作目录: {error}"))?
   171	        }
   172	    };
   173	    if !path.is_dir() {
   174	        return Err("工作目录必须是文件夹".to_string());
   175	    }
   176	    Ok(path)
   177	}

   179	/// Windows npm/pnpm installations expose Codex as a `.cmd` shim. Never pass
   180	/// assistant input through cmd.exe; resolve the fixed official JS entry and
   181	/// invoke Node with an argv array instead.
   182	#[cfg(target_os = "windows")]
   183	fn codex_assistant_command_for(executable: &Path) -> Result<Command, String> {
   184	    let is_batch = executable
   185	        .extension()
   186	        .and_then(|extension| extension.to_str())
   187	        .is_some_and(|extension| {
   188	            extension.eq_ignore_ascii_case("cmd") || extension.eq_ignore_ascii_case("bat")
   189	        });
   190	    if !is_batch {
   191	        return Ok(Command::new(executable));
   192	    }
   193	
   194	    let bin_dir = executable
   195	        .parent()
   196	        .ok_or_else(|| "Codex CLI 批处理入口没有父目录".to_string())?;
   197	    let entry = [
   198	        bin_dir.join("node_modules/@openai/codex/bin/codex.js"),
   199	        bin_dir.join("../@openai/codex/bin/codex.js"),
   200	    ]
   201	    .into_iter()
   202	    .find(|candidate| candidate.is_file())
   203	    .ok_or_else(|| {
   204	        format!(
   205	            "无法从 Codex CLI 入口 {} 定位官方 Node 启动文件",
   206	            executable.display()
   207	        )
   208	    })?;
   209	    let entry = super::misc::windows_shell_compatible_path(&entry);
   210	    let local_node = bin_dir.join("node.exe");
   211	    let node = if local_node.is_file() {
   212	        super::misc::windows_shell_compatible_path(&local_node)
   213	    } else {
   214	        let node = super::misc::resolve_tool_executable_for_gui("node");
   215	        super::misc::windows_shell_compatible_path(&node)
   216	    };
   217	    let mut command = Command::new(node);
   218	    command.arg(entry);
   219	    Ok(command)
   220	}

   224	    Ok(Command::new(executable))
   225	}

   248	fn request_key(id: &Value) -> Result<String, String> {
   249	    match id {
   250	        Value::String(_) | Value::Number(_) => {
   251	            serde_json::to_string(id).map_err(|error| format!("无效的 JSON-RPC 请求 ID: {error}"))
   252	        }
   253	        _ => Err("JSON-RPC 请求 ID 必须是字符串或数字".to_string()),
   254	    }
   255	}

   265	impl CodexAppServerSession {
   266	    fn write_message(&self, message: &Value) -> Result<(), String> {
   267	        let mut stdin = self
   268	            .stdin
   269	            .lock()
   270	            .map_err(|_| "Codex app-server 输入锁不可用".to_string())?;
   271	        serde_json::to_writer(&mut *stdin, message)
   272	            .map_err(|error| format!("无法编码 Codex app-server 消息: {error}"))?;
   273	        stdin
   274	            .write_all(b"\n")
   275	            .and_then(|_| stdin.flush())
   276	            .map_err(|error| format!("无法写入 Codex app-server: {error}"))
   277	    }
   278	
   279	    fn notify(&self, method: &str, params: Option<Value>) -> Result<(), String> {
   280	        let mut message = json!({ "method": method });
   281	        if let Some(params) = params {
   282	            message["params"] = params;
   283	        }
   284	        self.write_message(&message)
   285	    }
   286	
   287	    fn request(&self, method: &str, params: Value) -> Result<Value, String> {
   288	        let id = Value::from(self.next_request_id.fetch_add(1, Ordering::Relaxed));
   289	        let key = request_key(&id)?;
   290	        let (sender, receiver) = mpsc::channel();
   291	        self.pending_responses
   292	            .lock()
   293	            .map_err(|_| "Codex app-server 响应锁不可用".to_string())?
   294	            .insert(key.clone(), sender);
   295	
   296	        if let Err(error) = self.write_message(&json!({
   297	            "id": id,
   298	            "method": method,
   299	            "params": params,
   300	        })) {
   301	            if let Ok(mut pending) = self.pending_responses.lock() {
   302	                pending.remove(&key);
   303	            }
   304	            return Err(error);
   305	        }
   306	
   307	        match receiver.recv_timeout(RPC_TIMEOUT) {
   308	            Ok(result) => result,
   309	            Err(mpsc::RecvTimeoutError::Timeout) => {
   310	                if let Ok(mut pending) = self.pending_responses.lock() {
   311	                    pending.remove(&key);
   312	                }
   313	                Err(format!("Codex app-server 请求超时: {method}"))
   314	            }
   315	            Err(mpsc::RecvTimeoutError::Disconnected) => {
   316	                Err("Codex app-server 已断开连接".to_string())
   317	            }
   318	        }
   319	    }
   320	
   321	    fn fail_pending_responses(&self, error: &str) {
   322	        if let Ok(mut pending) = self.pending_responses.lock() {
   323	            for (_, sender) in pending.drain() {
   324	                let _ = sender.send(Err(error.to_string()));
   325	            }
   326	        }
   327	    }
   328	}

   348	fn is_high_risk_command(command: &str) -> bool {
   349	    let normalized = command
   350	        .to_ascii_lowercase()
   351	        .replace(['\r', '\n', '\t'], " ");
   352	    let padded = format!(" {normalized} ");
   353	    const HIGH_RISK_MARKERS: &[&str] = &[
   354	        " rm ",
   355	        " rm.exe ",
   356	        " rmdir ",
   357	        " del ",
   358	        " erase ",
   359	        " remove-item ",
   360	        " format ",
   361	        " diskpart ",
   362	        " clean all ",
   363	        " shutdown ",
   364	        " restart-computer ",
   365	        " stop-computer ",
   366	        " reboot ",
   367	        " reg delete ",
   368	        " sc delete ",
   369	        " net user ",
   370	        " bcdedit ",
   371	        " cipher /w ",
   372	        " invoke-expression ",
   373	        " iex ",
   374	        " encodedcommand ",
   375	        " --force ",
   376	        " -force ",
   377	        " reset --hard ",
   378	        " clean -fd ",
   379	        " clean -df ",
   380	    ];
   381	    HIGH_RISK_MARKERS
   382	        .iter()
   383	        .any(|marker| padded.contains(marker))
   384	        || normalized.contains("curl ")
   385	            && (normalized.contains("| sh") || normalized.contains("|sh"))
   386	        || normalized.contains("wget ")
   387	            && (normalized.contains("| sh") || normalized.contains("|sh"))
   388	        || normalized.contains(":(){:|:&};:")
   389	}

   391	fn advertised_string_decisions(params: &Value) -> Option<HashSet<String>> {
   392	    params
   393	        .get("availableDecisions")
   394	        .and_then(Value::as_array)
   395	        .map(|decisions| {
   396	            decisions
   397	                .iter()
   398	                .filter_map(Value::as_str)
   399	                .map(str::to_string)
   400	                .collect()
   401	        })
   402	}

   404	fn allowed_approval_decisions(method: &str, params: &Value, high_risk: bool) -> HashSet<String> {
   405	    let whitelist: HashSet<String> = match method {
   406	        COMMAND_APPROVAL_METHOD | FILE_APPROVAL_METHOD => {
   407	            ["accept", "acceptForSession", "decline", "cancel"]
   408	                .into_iter()
   409	                .map(str::to_string)
   410	                .collect()
   411	        }
   412	        _ => HashSet::new(),
   413	    };
   414	
   415	    let mut allowed: HashSet<String> = if let Some(advertised) = advertised_string_decisions(params)
   416	    {
   417	        whitelist.intersection(&advertised).cloned().collect()
   418	    } else {
   419	        ["accept", "cancel"]
   420	            .into_iter()
   421	            .map(str::to_string)
   422	            .collect()
   423	    };
   424	    if high_risk {
   425	        allowed.remove("acceptForSession");
   426	    }
   427	    allowed
   428	}

   430	fn ordered_decisions(decisions: &HashSet<String>) -> Vec<String> {
   431	    ["accept", "acceptForSession", "decline", "cancel"]
   432	        .into_iter()
   433	        .filter(|decision| decisions.contains(*decision))
   434	        .map(str::to_string)
   435	        .collect()
   436	}

   438	fn handle_server_request(session: &Arc<CodexAppServerSession>, message: &Value) {
   439	    let Some(method) = message.get("method").and_then(Value::as_str) else {
   440	        return;
   441	    };
   442	    let Some(request_id) = message.get("id").cloned() else {
   443	        return;
   444	    };
   445	    if request_key(&request_id).is_err() {
   446	        return;
   447	    };
   448	    let params = message.get("params").cloned().unwrap_or_else(|| json!({}));
   449	
   450	    if method != COMMAND_APPROVAL_METHOD && method != FILE_APPROVAL_METHOD {
   451	        let _ = session.write_message(&json!({
   452	            "id": request_id,
   453	            "error": { "code": -32601, "message": "Unsupported app-server request" }
   454	        }));
   455	        return;
   456	    }
   457	
   458	    let command = params.get("command").and_then(Value::as_str).unwrap_or("");
   459	    let high_risk = method == COMMAND_APPROVAL_METHOD && is_high_risk_command(command);
   460	    let available_decisions = allowed_approval_decisions(method, &params, high_risk);
   461	    if available_decisions.is_empty() {
   462	        let _ = session.write_message(&json!({
   463	            "id": request_id,
   464	            "result": { "decision": "cancel" }
   465	        }));
   466	        emit_event(
   467	            session.events.as_ref(),
   468	            CodexAssistantEvent::Stderr {
   469	                session_id: session.id.clone(),
   470	                message: "Codex 请求了当前 CC Switch 无法安全呈现的审批类型，已取消该操作。"
   471	                    .to_string(),
   472	            },
   473	        );
   474	        return;
   475	    }
   476	    let pending = PendingApproval {
   477	        request_id: request_id.clone(),
   478	        method: method.to_string(),
   479	        available_decisions: available_decisions.clone(),
   480	        high_risk,
   481	    };
   482	    let approval_id = Uuid::new_v4().to_string();
   483	    if let Ok(mut approvals) = session.pending_approvals.lock() {
   484	        approvals.insert(approval_id.clone(), pending);
   485	    } else {
   486	        let _ = session.write_message(&json!({
   487	            "id": request_id,
   488	            "result": { "decision": "cancel" }
   489	        }));
   490	        return;
   491	    }
   492	
   493	    let decisions = ordered_decisions(&available_decisions);
   494	    emit_event(
   495	        session.events.as_ref(),
   496	        CodexAssistantEvent::Approval {
   497	            session_id: session.id.clone(),
   498	            approval: CodexAssistantApproval {
   499	                id: approval_id,
   500	                approval_type: if method == COMMAND_APPROVAL_METHOD {
   501	                    "command"
   502	                } else {
   503	                    "fileChange"
   504	                }
   505	                .to_string(),
   506	                command: params
   507	                    .get("command")
   508	                    .and_then(Value::as_str)
   509	                    .map(str::to_string),
   510	                cwd: params
   511	                    .get("cwd")
   512	                    .and_then(Value::as_str)
   513	                    .map(str::to_string),
   514	                reason: params
   515	                    .get("reason")
   516	                    .and_then(Value::as_str)
   517	                    .map(str::to_string),
   518	                network_host: params
   519	                    .pointer("/networkApprovalContext/host")
   520	                    .and_then(Value::as_str)
   521	                    .map(str::to_string),
   522	                grant_root: params
   523	                    .get("grantRoot")
   524	                    .and_then(Value::as_str)
   525	                    .map(str::to_string),
   526	                allow_for_session: decisions
   527	                    .iter()
   528	                    .any(|decision| decision == "acceptForSession"),
   529	                available_decisions: decisions,
   530	            },
   531	        },
   532	    );
   533	}

   535	fn handle_protocol_message(session: &Arc<CodexAppServerSession>, message: Value) {
   536	    if let Some(id) = message.get("id") {
   537	        if message.get("method").is_some() {
   538	            handle_server_request(session, &message);
   539	            return;
   540	        }
   541	        if let Ok(key) = request_key(id) {
   542	            let sender = session
   543	                .pending_responses
   544	                .lock()
   545	                .ok()
   546	                .and_then(|mut pending| pending.remove(&key));
   547	            if let Some(sender) = sender {
   548	                let result = if message.get("error").is_some() {
   549	                    Err(response_error(&message))
   550	                } else {
   551	                    Ok(message.get("result").cloned().unwrap_or(Value::Null))
   552	                };
   553	                let _ = sender.send(result);
   554	            }
   555	        }
   556	        return;
   557	    }
   558	
   559	    let Some(method) = message.get("method").and_then(Value::as_str) else {
   560	        return;
   561	    };
   562	    let params = message.get("params").cloned().unwrap_or(Value::Null);
   563	    match method {
   564	        "item/agentMessage/delta" => {
   565	            if let Some(delta) = params.get("delta").and_then(Value::as_str) {
   566	                emit_event(
   567	                    session.events.as_ref(),
   568	                    CodexAssistantEvent::Message {
   569	                        session_id: session.id.clone(),
   570	                        message: delta.to_string(),
   571	                    },
   572	                );
   573	            }
   574	        }
   575	        "item/commandExecution/outputDelta" | "item/commandExecution/terminalInteraction" => {
   576	            if let Some(delta) = params.get("delta").and_then(Value::as_str) {
   577	                emit_event(
   578	                    session.events.as_ref(),
   579	                    CodexAssistantEvent::Log {
   580	                        session_id: session.id.clone(),
   581	                        message: delta.to_string(),
   582	                    },
   583	                );
   584	            }
   585	        }
   586	        "item/fileChange/outputDelta" | "item/fileChange/patchUpdated" => {
   587	            emit_event(
   588	                session.events.as_ref(),
   589	                CodexAssistantEvent::Log {
   590	                    session_id: session.id.clone(),
   591	                    message: params.to_string(),
   592	                },
   593	            );
   594	        }
   595	        "turn/started" => {
   596	            session.turn_active.store(true, Ordering::Release);
   597	            if let Some(turn_id) = params
   598	                .get("turnId")
   599	                .or_else(|| params.pointer("/turn/id"))
   600	                .and_then(Value::as_str)
   601	            {
   602	                if let Ok(mut active) = session.active_turn_id.lock() {
   603	                    *active = Some(turn_id.to_string());
   604	                }
   605	            }
   606	            emit_event(
   607	                session.events.as_ref(),
   608	                CodexAssistantEvent::Started {
   609	                    session_id: session.id.clone(),
   610	                },
   611	            );
   612	        }
   613	        "turn/completed" => {
   614	            session.turn_active.store(false, Ordering::Release);
   615	            if let Ok(mut active) = session.active_turn_id.lock() {
   616	                *active = None;
   617	            }
   618	            if let Ok(mut approvals) = session.pending_approvals.lock() {
   619	                approvals.clear();
   620	            }
   621	            let status = params.pointer("/turn/status").and_then(Value::as_str);
   622	            let success = matches!(status, Some("completed"));
   623	            let cancelled = matches!(status, Some("interrupted" | "cancelled"));
   624	            let error = params
   625	                .pointer("/turn/error/message")
   626	                .or_else(|| params.pointer("/turn/error"))
   627	                .and_then(Value::as_str)
   628	                .map(str::to_string)
   629	                .or_else(|| {
   630	                    (!success && !cancelled).then(|| status.unwrap_or("Codex 回合失败").to_string())
   631	                });
   632	            emit_event(
   633	                session.events.as_ref(),
   634	                CodexAssistantEvent::Finished {
   635	                    session_id: session.id.clone(),
   636	                    success,
   637	                    cancelled,
   638	                    message: error,
   639	                },
   640	            );
   641	        }
   642	        "error" => {
   643	            let error = params
   644	                .pointer("/error/message")
   645	                .and_then(Value::as_str)
   646	                .unwrap_or("Codex app-server 报告错误")
   647	                .to_string();
   648	            emit_event(
   649	                session.events.as_ref(),
   650	                CodexAssistantEvent::Stderr {
   651	                    session_id: session.id.clone(),
   652	                    message: error,
   653	                },
   654	            );
   655	        }
   656	        _ => {}
   657	    }
   658	}

   660	fn start_protocol_readers(
   661	    session: Arc<CodexAppServerSession>,
   662	    stdout: impl std::io::Read + Send + 'static,
   663	    stderr: impl std::io::Read + Send + 'static,
   664	) {
   665	    let stdout_session = session.clone();
   666	    thread::spawn(move || {
   667	        for line in BufReader::new(stdout).lines() {
   668	            match line {
   669	                Ok(line) if !line.trim().is_empty() => match serde_json::from_str::<Value>(&line) {
   670	                    Ok(message) => handle_protocol_message(&stdout_session, message),
   671	                    Err(error) => emit_event(
   672	                        stdout_session.events.as_ref(),
   673	                        CodexAssistantEvent::Stderr {
   674	                            session_id: stdout_session.id.clone(),
   675	                            message: format!("无法解析 Codex app-server 消息: {error}"),
   676	                        },
   677	                    ),
   678	                },
   679	                Ok(_) => {}
   680	                Err(error) => {
   681	                    emit_event(
   682	                        stdout_session.events.as_ref(),
   683	                        CodexAssistantEvent::Stderr {
   684	                            session_id: stdout_session.id.clone(),
   685	                            message: format!("读取 Codex app-server 输出失败: {error}"),
   686	                        },
   687	                    );
   688	                    break;
   689	                }
   690	            }
   691	        }
   692	        stdout_session.fail_pending_responses("Codex app-server 已断开连接");
   693	    });
   694	
   695	    let stderr_session = session.clone();
   696	    thread::spawn(move || {
   697	        for line in BufReader::new(stderr).lines().map_while(Result::ok) {
   698	            emit_event(
   699	                stderr_session.events.as_ref(),
   700	                CodexAssistantEvent::Stderr {
   701	                    session_id: stderr_session.id.clone(),
   702	                    message: line,
   703	                },
   704	            );
   705	        }
   706	    });
   707	
   708	    thread::spawn(move || {
   709	        loop {
   710	            let status = match session.child.lock() {
   711	                Ok(mut child) => child.try_wait(),
   712	                Err(_) => return,
   713	            };
   714	            match status {
   715	                Ok(Some(_)) => break,
   716	                Ok(None) => thread::sleep(Duration::from_millis(100)),
   717	                Err(_) => break,
   718	            }
   719	        }
   720	        let turn_was_active = session.turn_active.swap(false, Ordering::AcqRel);
   721	        if let Ok(mut approvals) = session.pending_approvals.lock() {
   722	            approvals.clear();
   723	        }
   724	        session.fail_pending_responses("Codex app-server 已退出");
   725	        if !session.closing.load(Ordering::Acquire) {
   726	            let message = if turn_was_active {
   727	                "Codex app-server 在回复过程中意外退出"
   728	            } else {
   729	                "Codex app-server 已意外断开，请重新发送消息以建立新会话"
   730	            };
   731	            log::error!(
   732	                "Codex assistant session {} disconnected: {message}",
   733	                session.id
   734	            );
   735	            emit_event(
   736	                session.events.as_ref(),
   737	                CodexAssistantEvent::Disconnected {
   738	                    session_id: session.id.clone(),
   739	                    message: message.to_string(),
   740	                },
   741	            );
   742	        }
   743	        if let Ok(mut sessions) = SESSIONS.lock() {
   744	            sessions.remove(&session.id);
   745	        }
   746	    });
   747	}

   763	fn spawn_app_server(app: AppHandle) -> Result<Arc<CodexAppServerSession>, String> {
   764	    let executable = super::misc::resolve_tool_executable_for_gui("codex");
   765	    let mut command = codex_assistant_command_for(&executable)?;
   766	    let codex_home = codex_assistant_home();
   767	    fs::create_dir_all(&codex_home).map_err(|error| format!("无法创建 Codex 配置目录: {error}"))?;
   768	    configure_app_server_command(&mut command, &codex_home);
   769	    #[cfg(target_os = "windows")]
   770	    {
   771	        use std::os::windows::process::CommandExt;
   772	        command.creation_flags(0x08000000);
   773	    }
   774	
   775	    let mut child = command
   776	        .spawn()
   777	        .map_err(|error| format!("无法启动 Codex app-server: {error}"))?;
   778	    let stdin = child
   779	        .stdin
   780	        .take()
   781	        .ok_or_else(|| "无法打开 Codex app-server 输入".to_string())?;
   782	    let stdout = child
   783	        .stdout
   784	        .take()
   785	        .ok_or_else(|| "无法读取 Codex app-server 输出".to_string())?;
   786	    let stderr = child
   787	        .stderr
   788	        .take()
   789	        .ok_or_else(|| "无法读取 Codex app-server 错误输出".to_string())?;
   790	    let session = Arc::new(CodexAppServerSession {
   791	        id: Uuid::new_v4().to_string(),
   792	        events: Arc::new(TauriEventSink(app)),
   793	        child: Mutex::new(Box::new(child)),
   794	        stdin: Mutex::new(Box::new(stdin)),
   795	        next_request_id: AtomicU64::new(1),
   796	        pending_responses: Mutex::new(HashMap::new()),
   797	        pending_approvals: Mutex::new(HashMap::new()),
   798	        thread_id: Mutex::new(None),
   799	        active_turn_id: Mutex::new(None),
   800	        turn_active: AtomicBool::new(false),
   801	        closing: AtomicBool::new(false),
   802	    });
   803	    SESSIONS
   804	        .lock()
   805	        .map_err(|_| "Codex 会话状态锁不可用".to_string())?
   806	        .insert(session.id.clone(), session.clone());
   807	    start_protocol_readers(session.clone(), stdout, stderr);
   808	    Ok(session)
   809	}

   824	    }
   825	    let _ = child.kill();
   826	}

   874	    let input = validate_input(&message)?;
   875	    let session = get_session(&session_id)?;
   876	    if session
   877	        .turn_active
   878	        .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
   879	        .is_err()
   880	    {
   881	        return Err("当前 Codex 回合尚未结束".to_string());
   882	    }
   883	    let result = (|| {
   884	        let thread_id = session
   885	            .thread_id
   886	            .lock()
   887	            .map_err(|_| "Codex thread 状态锁不可用".to_string())?
   888	            .clone()
   889	            .ok_or_else(|| "Codex thread 尚未初始化".to_string())?;
   890	        let result = session.request(
   891	            "turn/start",
   892	            json!({
   893	                "threadId": thread_id,
   894	                "input": [{ "type": "text", "text": input }]
   895	            }),
   896	        )?;
   897	        let turn_id = extract_id(&result, "/turn/id", "turn id")?;
   898	        if session.turn_active.load(Ordering::Acquire) {
   899	            *session
   900	                .active_turn_id
   901	                .lock()
   902	                .map_err(|_| "Codex turn 状态锁不可用".to_string())? = Some(turn_id);
   903	        }
   904	        Ok(())
   905	    })();
   906	    if result.is_err() {
   907	        session.turn_active.store(false, Ordering::Release);
   908	    }
   909	    result
   910	}

   913	pub fn respond_codex_assistant_approval(
   914	    session_id: String,
   915	    approval_id: String,
   916	    decision: String,
   917	) -> Result<bool, String> {
   918	    let session = get_session(&session_id)?;
   919	    let decision = decision.trim();
   920	    let pending = {
   921	        let mut approvals = session
   922	            .pending_approvals
   923	            .lock()
   924	            .map_err(|_| "Codex 审批状态锁不可用".to_string())?;
   925	        let pending = approvals
   926	            .get(&approval_id)
   927	            .cloned()
   928	            .ok_or_else(|| "审批请求不存在或已处理".to_string())?;
   929	        if !pending.available_decisions.contains(decision) {
   930	            return Err(format!("当前审批不支持 decision: {decision}"));
   931	        }
   932	        if decision == "acceptForSession"
   933	            && pending.method == COMMAND_APPROVAL_METHOD
   934	            && pending.high_risk
   935	        {
   936	            return Err("高风险命令不能在本会话中持续授权".to_string());
   937	        }
   938	        approvals.remove(&approval_id);
   939	        pending
   940	    };
   941	    if let Err(error) = session.write_message(&json!({
   942	        "id": pending.request_id,
   943	        "result": { "decision": decision }
   944	    })) {
   945	        if let Ok(mut approvals) = session.pending_approvals.lock() {
   946	            approvals.insert(approval_id, pending);
   947	        }
   948	        return Err(error);
   949	    }
   950	    Ok(true)
   951	}
