     2	import type { PointerEvent as ReactPointerEvent } from "react";
     3	import {
     4	  Bot,
     5	  Check,

     7	  CircleAlert,

    28	type CodexRuntime = {
    29	  version: string | null;
    30	  installed_but_broken: boolean;
    31	};
    32	
    33	type FloatingPosition = {
    34	  x: number;
    35	  y: number;
    36	};
    37	
    38	const FLOATING_BUTTON_SIZE = 34;
    39	const FLOATING_BUTTON_MARGIN = 16;
    40	const FLOATING_POSITION_STORAGE_KEY = "cc-switch-codex-assistant-position";
    41	const MAX_VISIBLE_LOG_LINES = 180;
    42	
    43	const clampFloatingPosition = (
    44	  position: FloatingPosition,
    45	): FloatingPosition => ({
    46	  x: Math.min(
    47	    Math.max(FLOATING_BUTTON_MARGIN, position.x),
    48	    Math.max(
    49	      FLOATING_BUTTON_MARGIN,
    50	      window.innerWidth - FLOATING_BUTTON_SIZE - FLOATING_BUTTON_MARGIN,
    51	    ),
    52	  ),
    53	  y: Math.min(
    54	    Math.max(FLOATING_BUTTON_MARGIN, position.y),
    55	    Math.max(
    56	      FLOATING_BUTTON_MARGIN,
    57	      window.innerHeight - FLOATING_BUTTON_SIZE - FLOATING_BUTTON_MARGIN,
    58	    ),
    59	  ),
    60	});
    61	
    62	const defaultFloatingPosition = (): FloatingPosition =>
    63	  clampFloatingPosition({
    64	    x: window.innerWidth - FLOATING_BUTTON_SIZE - FLOATING_BUTTON_MARGIN,
    65	    y: window.innerHeight / 2 - FLOATING_BUTTON_SIZE / 2,
    66	  });

    68	export interface CodexAssistantInstallIntent {
    69	  tool: string;
    70	  appName: string;
    71	  supportsCustomLocation?: boolean;
    72	  supportsVersionPin?: boolean;
    73	}
    74	
    75	interface CodexAssistantDockProps {
    76	  providerReady: boolean;
    77	  onOpenCodexConfiguration: () => void;
    78	  onDismiss?: () => void;
    79	  onInstallationCompleted?: (tool: string) => void;
    80	  installIntent?: CodexAssistantInstallIntent | null;
    81	  /** Incremented by external entry points such as “Add custom Agent”. */
    82	  openRequestId?: number;
    83	}
    84	
    85	/** A persistent Codex app-server conversation running in full-auto mode. */
    86	export function CodexAssistantDock({
    87	  providerReady,
    88	  onOpenCodexConfiguration,
    89	  onDismiss,
    90	  onInstallationCompleted,
    91	  installIntent,
    92	  openRequestId = 0,
    93	}: CodexAssistantDockProps) {
    94	  const { t } = useTranslation();
    95	  const webBridgeActive = isCodexAssistantWebBridgeActive();
    96	  const [open, setOpen] = useState(false);
    97	  const [showFloatingGreeting, setShowFloatingGreeting] = useState(true);
    98	  const [chatMessages, setChatMessages] = useState<CodexAssistantChatTurn[]>(
    99	    [],
   100	  );
   101	  const [runtime, setRuntime] = useState<CodexRuntime | null>(null);
   102	  const [checking, setChecking] = useState(false);
   103	  const [installing, setInstalling] = useState(false);
   104	  const [request, setRequest] = useState("");
   105	  const [logs, setLogs] = useState<string[]>([]);
   106	  const [runError, setRunError] = useState<string | null>(null);
   107	  const [startingSession, setStartingSession] = useState(false);
   108	  const [running, setRunning] = useState(false);
   109	  const [stopping, setStopping] = useState(false);
   110	  const [sessionId, setSessionId] = useState<string | null>(null);
   111	  const [floatingPosition, setFloatingPosition] =
   112	    useState<FloatingPosition | null>(null);
   113	  const floatingDockRef = useRef<HTMLDivElement>(null);
   114	  const latestFloatingPositionRef = useRef<FloatingPosition | null>(null);
   115	  const pendingFloatingPositionRef = useRef<FloatingPosition | null>(null);
   116	  const dragAnimationFrameRef = useRef<number | null>(null);
   117	  const dragStateRef = useRef<{
   118	    pointerId: number;
   119	    startClientX: number;
   120	    startClientY: number;
   121	    startPosition: FloatingPosition;
   122	    moved: boolean;
   123	  } | null>(null);
   124	  const suppressOpenRef = useRef(false);
   125	  const sessionIdRef = useRef<string | null>(null);
   126	  const streamAssistantMessageRef = useRef(false);
   127	  const lastHandledOpenRequestRef = useRef(0);
   128	
   129	  const appendLog = useCallback((line: string) => {
   130	    setLogs((current) => [...current, line].slice(-MAX_VISIBLE_LOG_LINES));
   131	  }, []);
   132	
   133	  useEffect(() => {
   134	    const storedPosition = localStorage.getItem(FLOATING_POSITION_STORAGE_KEY);
   135	    if (storedPosition) {
   136	      try {
   137	        const parsed = JSON.parse(storedPosition) as Partial<FloatingPosition>;
   138	        if (typeof parsed.x === "number" && typeof parsed.y === "number") {
   139	          const next = clampFloatingPosition(parsed as FloatingPosition);
   140	          latestFloatingPositionRef.current = next;
   141	          setFloatingPosition(next);
   142	          return;
   143	        }
   144	      } catch {
   145	        // Ignore stale stored coordinates.
   146	      }
   147	    }
   148	    const next = defaultFloatingPosition();
   149	    latestFloatingPositionRef.current = next;
   150	    setFloatingPosition(next);
   151	  }, []);
   152	
   153	  useEffect(() => {
   154	    if (
   155	      openRequestId <= 0 ||
   156	      openRequestId === lastHandledOpenRequestRef.current
   157	    ) {
   158	      return;
   159	    }
   160	    lastHandledOpenRequestRef.current = openRequestId;
   161	    if (installIntent) {
   162	      setRequest(
   163	        t("codexAssistant.installRequest", { app: installIntent.appName }),
   164	      );
   165	    }
   166	    setShowFloatingGreeting(false);
   167	    setOpen(true);
   168	  }, [installIntent, openRequestId, t]);
   169	
   170	  useEffect(() => {
   171	    const keepFloatingButtonVisible = () => {
   172	      const next = clampFloatingPosition(
   173	        latestFloatingPositionRef.current ?? defaultFloatingPosition(),
   174	      );
   175	      latestFloatingPositionRef.current = next;
   176	      setFloatingPosition(next);
   177	    };
   178	    window.addEventListener("resize", keepFloatingButtonVisible);
   179	    return () =>
   180	      window.removeEventListener("resize", keepFloatingButtonVisible);
   181	  }, []);
   182	
   183	  useEffect(
   184	    () => () => {
   185	      if (dragAnimationFrameRef.current !== null) {
   186	        cancelAnimationFrame(dragAnimationFrameRef.current);
   187	      }
   188	      const activeSessionId = sessionIdRef.current;
   189	      sessionIdRef.current = null;
   190	      if (activeSessionId) {
   191	        void settingsApi.closeCodexAssistantSession(activeSessionId);
   192	      }
   193	    },
   194	    [],
   195	  );
   196	
   197	  const refreshRuntime = useCallback(async () => {
   198	    setChecking(true);
   199	    try {
   200	      const [next] = await settingsApi.getToolVersions(["codex"]);
   201	      setRuntime(next ?? null);
   202	    } catch (error) {
   203	      toast.error(t("codexAssistant.runtimeCheckFailed"), {
   204	        description: extractErrorMessage(error) || undefined,
   205	      });
   206	    } finally {
   207	      setChecking(false);
   208	    }
   209	  }, [t]);
   210	
   211	  useEffect(() => {
   212	    let dispose = false;
   213	    let unlisten: (() => void) | undefined;
   214	
   215	    void settingsApi
   216	      .onCodexAssistantEvent((event) => {
   217	        if (event.sessionId !== sessionIdRef.current) return;
   218	
   219	        switch (event.kind) {
   220	          case "started":
   221	            streamAssistantMessageRef.current = false;
   222	            setRunning(true);
   223	            return;
   224	          case "log":
   225	            if (event.message) appendLog(event.message);
   226	            return;
   227	          case "stderr":
   228	            if (event.message) appendLog(`stderr · ${event.message}`);
   229	            return;
   230	          case "disconnected":
   231	            sessionIdRef.current = null;
   232	            setSessionId(null);
   233	            streamAssistantMessageRef.current = false;
   234	            setRunning(false);
   235	            setStopping(false);
   236	            setApprovals([]);
   237	            setRespondingApprovalIds(new Set());
   238	            setRunError(event.message);
   239	            toast.error(t("codexAssistant.runFailed"), {
   240	              description: event.message,
   241	            });
   242	            return;
   243	          case "message":
   244	            if (event.message) {
   245	              setChatMessages((current) => {
   246	                if (streamAssistantMessageRef.current) {
   247	                  const last = current.at(-1);
   248	                  if (last?.role === "assistant") {
   249	                    return [
   250	                      ...current.slice(0, -1),
   251	                      { ...last, content: last.content + event.message },
   252	                    ];
   253	                  }
   254	                }
   255	                streamAssistantMessageRef.current = true;
   256	                return [
   257	                  ...current,
   258	                  { role: "assistant", content: event.message },
   259	                ];
   260	              });
   261	            }
   262	            return;
   263	          case "approval":
   264	            setApprovals((current) => [
   265	              ...current.filter((item) => item.id !== event.approval.id),
   266	              event.approval,
   267	            ]);
   268	            return;
   269	          case "finished":
   270	            streamAssistantMessageRef.current = false;
   271	            setRunning(false);
   272	            setStopping(false);
   273	            setApprovals([]);
   274	            setRespondingApprovalIds(new Set());
   275	            if (!event.success && !event.cancelled) {
   276	              const message = event.message || t("codexAssistant.runFailed");
   277	              setRunError(message);
   278	              toast.error(t("codexAssistant.runFailed"), {
   279	                description: message,
   280	              });
   281	            }
   282	        }
   283	      })
   284	      .then((listener) => {
   285	        if (dispose) listener();
   286	        else unlisten = listener;
   287	      });
   288	
   289	    return () => {
   290	      dispose = true;
   291	      unlisten?.();
   292	    };
   293	  }, [appendLog, t]);
   294	
   295	  useEffect(() => {
   296	    if (open) void refreshRuntime();
   297	  }, [open, refreshRuntime]);
   298	
   299	  const installCodex = async () => {
   300	    setInstalling(true);
   301	    try {
   302	      await settingsApi.runToolLifecycleAction(["codex"], "install");
   303	      await refreshRuntime();
   304	      onInstallationCompleted?.("codex");
   305	      toast.success(t("codexAssistant.runtimeInstalled"));
   306	    } catch (error) {
   307	      toast.error(t("codexAssistant.runtimeInstallFailed"), {
   308	        description: extractErrorMessage(error) || undefined,
   309	      });
   310	    } finally {
   311	      setInstalling(false);
   312	    }
   313	  };
   314	
   315	  const cliReady = Boolean(runtime?.version && !runtime?.installed_but_broken);
   316	  const effectiveCliReady = cliReady || webBridgeActive;
   317	  const effectiveProviderReady = providerReady || webBridgeActive;
   318	  const ready = effectiveCliReady && effectiveProviderReady;
   319	  const blockedReason = !effectiveCliReady
   320	    ? t("codexAssistant.cliRequired")
   321	    : !effectiveProviderReady
   322	      ? t("codexAssistant.providerRequired")
   323	      : null;
   324	
   325	  const ensureSession = async (): Promise<string> => {
   326	    if (sessionIdRef.current) return sessionIdRef.current;
   327	    setStartingSession(true);
   328	    try {
   329	      const nextSessionId = await settingsApi.startCodexAssistantSession();
   330	      sessionIdRef.current = nextSessionId;
   331	      setSessionId(nextSessionId);
   332	      return nextSessionId;
   333	    } finally {
   334	      setStartingSession(false);
   335	    }
   336	  };
   337	
   338	  const sendMessage = async () => {
   339	    const text = request.trim();
   340	    if (!text || running || startingSession) return;
   341	
   342	    setRequest("");
   343	    setRunError(null);
   344	    setLogs([]);
   345	    streamAssistantMessageRef.current = false;
   346	    setChatMessages((current) => [...current, { role: "user", content: text }]);
   347	    try {
   348	      const activeSessionId = await ensureSession();
   349	      setRunning(true);
   350	      await settingsApi.sendCodexAssistantMessage(activeSessionId, text);
   351	    } catch (error) {
   352	      setRunning(false);
   353	      const message =
   354	        extractErrorMessage(error) || t("codexAssistant.runFailed");
   355	      setRunError(message);
   356	      toast.error(t("codexAssistant.runFailed"), { description: message });
   357	    }
   358	  };
   359	
   360	  const respondToApproval = async (
   361	    approvalId: string,
   362	    decision: CodexAssistantApprovalDecision,
   363	  ) => {
   364	    const activeSessionId = sessionIdRef.current;
   365	    if (!activeSessionId || respondingApprovalIds.has(approvalId)) return;
   366	    setRespondingApprovalIds((current) => new Set(current).add(approvalId));
   367	    try {
   368	      await settingsApi.respondCodexAssistantApproval(
   369	        activeSessionId,
   370	        approvalId,
   371	        decision,
   372	      );
   373	      setApprovals((current) =>
   374	        current.filter((approval) => approval.id !== approvalId),
   375	      );
   376	    } catch (error) {
   377	      toast.error(
   378	        t("codexAssistant.approvalFailed", {
   379	          defaultValue: "Could not send the approval decision",
   380	        }),
   381	        { description: extractErrorMessage(error) || undefined },
   382	      );
   383	    } finally {
   384	      setRespondingApprovalIds((current) => {
   385	        const next = new Set(current);
   386	        next.delete(approvalId);
   387	        return next;
   388	      });
   389	    }
   390	  };
   391	
   392	  const cancelRun = async () => {
   393	    const activeSessionId = sessionIdRef.current;
   394	    if (!activeSessionId || stopping) return;
   395	    setStopping(true);
   396	    try {
   397	      await settingsApi.cancelCodexAssistantRun(activeSessionId);
   398	      appendLog(t("codexAssistant.cancelling"));
   399	    } catch (error) {
   400	      setStopping(false);
   401	      toast.error(t("codexAssistant.cancelFailed"), {
   402	        description: extractErrorMessage(error) || undefined,
   403	      });
   404	    }
   405	  };
   406	
   407	  const closeSession = () => {
   408	    const activeSessionId = sessionIdRef.current;
   409	    sessionIdRef.current = null;
   410	    setSessionId(null);
   411	    streamAssistantMessageRef.current = false;
   412	    setRunning(false);
   413	    setStopping(false);
   414	    setApprovals([]);
   415	    setRespondingApprovalIds(new Set());
   416	    if (activeSessionId) {
   417	      void settingsApi
   418	        .closeCodexAssistantSession(activeSessionId)
   419	        .catch(() => undefined);
   420	    }
   421	  };
   422	
   423	  const persistFloatingPosition = (position: FloatingPosition) => {
   424	    localStorage.setItem(
   425	      FLOATING_POSITION_STORAGE_KEY,
   426	      JSON.stringify(position),
   427	    );
   428	  };
   429	
   430	  const moveFloatingButton = (position: FloatingPosition) => {
   431	    latestFloatingPositionRef.current = position;
   432	    pendingFloatingPositionRef.current = position;
   433	    if (dragAnimationFrameRef.current !== null) return;
   434	
   435	    dragAnimationFrameRef.current = requestAnimationFrame(() => {
   436	      const next = pendingFloatingPositionRef.current;
   437	      const dock = floatingDockRef.current;
   438	      if (next && dock) {
   439	        dock.style.transform = `translate3d(${next.x}px, ${next.y}px, 0)`;
   440	      }
   441	      dragAnimationFrameRef.current = null;
   442	    });
   443	  };
   444	
   445	  const handlePointerDown = (event: ReactPointerEvent<HTMLButtonElement>) => {
   446	    if (event.button !== 0) return;
   447	    event.preventDefault();
   448	    const current =
   449	      latestFloatingPositionRef.current ??
   450	      floatingPosition ??
   451	      defaultFloatingPosition();
   452	    event.currentTarget.setPointerCapture(event.pointerId);
   453	    dragStateRef.current = {
   454	      pointerId: event.pointerId,
   455	      startClientX: event.clientX,
   456	      startClientY: event.clientY,
   457	      startPosition: current,
   458	      moved: false,
   459	    };
   460	  };
   461	
   462	  const handlePointerMove = (event: ReactPointerEvent<HTMLButtonElement>) => {
   463	    const drag = dragStateRef.current;
   464	    if (!drag || drag.pointerId !== event.pointerId) return;
   465	    event.preventDefault();
   466	    const deltaX = event.clientX - drag.startClientX;
   467	    const deltaY = event.clientY - drag.startClientY;
   468	    if (Math.abs(deltaX) + Math.abs(deltaY) > 4) drag.moved = true;
   469	    if (!drag.moved) return;
   470	    moveFloatingButton(
   471	      clampFloatingPosition({
   472	        x: drag.startPosition.x + deltaX,
   473	        y: drag.startPosition.y + deltaY,
   474	      }),
   475	    );
   476	  };
   477	
   478	  const finishDrag = (event: ReactPointerEvent<HTMLButtonElement>) => {
   479	    const drag = dragStateRef.current;
   480	    if (!drag || drag.pointerId !== event.pointerId) return;
   481	    dragStateRef.current = null;
   482	    if (!drag.moved) return;
   483	    const nextPosition = clampFloatingPosition({
   484	      x: drag.startPosition.x + event.clientX - drag.startClientX,
   485	      y: drag.startPosition.y + event.clientY - drag.startClientY,
   486	    });
   487	    moveFloatingButton(nextPosition);
   488	    setFloatingPosition(nextPosition);
   489	    persistFloatingPosition(nextPosition);
   490	    suppressOpenRef.current = true;
   491	  };
   492	
   493	  const openAssistant = () => {
   494	    if (suppressOpenRef.current) {
   495	      suppressOpenRef.current = false;
   496	      return;
   497	    }
   498	    setShowFloatingGreeting(false);
   499	    setOpen(true);
   500	  };
   501	
   502	  const closeAssistant = () => {
   503	    setOpen(false);
   504	    closeSession();
   505	    onDismiss?.();
   506	  };
   507	
   508	  useEffect(() => {
   509	    if (!open) return;
   510	    const handleKeyDown = (event: KeyboardEvent) => {
   511	      if (event.key === "Escape") {
   512	        event.preventDefault();
   513	        closeAssistant();
   514	      }
   515	    };
   516	    window.addEventListener("keydown", handleKeyDown);
   517	    return () => window.removeEventListener("keydown", handleKeyDown);
   518	  }, [open, onDismiss]);
   519	
   520	  return (
   521	    <>
   522	      <div
   523	        ref={floatingDockRef}
   524	        style={
   525	          floatingPosition
   526	            ? {
   527	                left: 0,
   528	                top: 0,
   529	                transform: `translate3d(${floatingPosition.x}px, ${floatingPosition.y}px, 0)`,
   530	              }
   531	            : { right: FLOATING_BUTTON_MARGIN, top: "calc(50% - 17px)" }
   532	        }
   533	        className="fixed z-[120] isolate will-change-transform"
   534	      >
   535	        {showFloatingGreeting && (
   536	          <div className="pointer-events-none absolute right-[calc(100%+10px)] top-1/2 hidden w-56 -translate-y-1/2 rounded-2xl border border-violet-500/25 bg-background/75 py-2 pl-3 pr-8 text-left text-xs leading-relaxed text-foreground/75 opacity-90 shadow-[0_8px_24px_-16px_rgba(76,29,149,0.45)] backdrop-blur-[2px] sm:block">
   537	            <span>{t("codexAssistant.floatingGreeting")}</span>
   538	            <button
   539	              type="button"
   540	              className="pointer-events-auto absolute right-1.5 top-1.5 flex h-5 w-5 items-center justify-center rounded-full text-muted-foreground/70 transition-colors hover:bg-violet-500/10 hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-500/40"
   541	              onClick={() => setShowFloatingGreeting(false)}
   542	              aria-label={t("codexAssistant.dismissGreeting")}
   543	              title={t("codexAssistant.dismissGreeting")}
   544	            >
   545	              <X className="h-3 w-3" />
   546	            </button>
   547	          </div>
   548	        )}
   549	        <button
   550	          type="button"
   551	          onClick={openAssistant}
   552	          onPointerDown={handlePointerDown}
   553	          onPointerMove={handlePointerMove}
   554	          onPointerUp={finishDrag}
   555	          onPointerCancel={finishDrag}
   556	          className="flex h-[34px] w-[34px] cursor-grab touch-none select-none items-center justify-center rounded-full border border-violet-500/35 bg-background text-violet-600 shadow-[0_10px_22px_-10px_rgba(109,40,217,0.7)] backdrop-blur transition-[background-color,box-shadow] hover:bg-violet-500/10 active:cursor-grabbing dark:text-violet-300"
   557	          aria-label={t("codexAssistant.open")}
   558	          title={t("codexAssistant.open")}
   559	        >
   560	          <CodexIcon size={19} className="dark:invert-0" />
   561	        </button>
   562	      </div>
   563	
   564	      <aside
   565	        aria-hidden={!open}
   566	        className={cn(
   567	          "fixed inset-y-0 right-0 z-[130] flex w-[390px] max-w-[calc(100vw-24px)] flex-col border-l border-border bg-background shadow-[-18px_0_44px_-24px_rgba(15,23,42,0.45)] transition-transform duration-300 ease-out",
   568	          open ? "translate-x-0" : "translate-x-full",
   569	        )}
   570	      >
   571	        <header className="flex items-center justify-between border-b border-border px-5 py-4">
   572	          <div className="flex items-center gap-3">
   573	            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-gradient-to-br from-violet-500 to-indigo-500 text-white shadow-sm">
   574	              <CodexIcon size={17} className="dark:invert-0" />
   575	            </span>
   576	            <div>
   577	              <h2 className="text-sm font-semibold">
   578	                {t("codexAssistant.title")}
   579	              </h2>
   580	              <p className="text-xs text-muted-foreground">
   581	                {t("codexAssistant.subtitle")}
   582	              </p>
   583	            </div>
   584	          </div>
   585	          <Button
   586	            variant="ghost"

   595	
   596	        <div className="flex-1 space-y-5 overflow-y-auto p-5">

   600	                <Bot className="h-4 w-4" />
   601	              </span>

   605	                  {t("codexAssistant.intro")}
   606	                </p>
   607	              </div>
   608	            </div>
   609	          </div>
   610	
   611	          <section className="space-y-2">
   612	            <p className="px-1 text-xs font-medium uppercase tracking-[0.12em] text-muted-foreground">
   613	              {t("codexAssistant.readiness")}
   614	            </p>
   615	            <button
   616	              type="button"
   617	              onClick={() => !cliReady && void installCodex()}
   618	              disabled={installing || checking || cliReady}
   619	              className="flex w-full items-center gap-3 rounded-xl border border-border bg-card px-3 py-3 text-left transition hover:border-violet-500/30 disabled:cursor-default"
   620	            >
   621	              <span

   623	                  "flex h-8 w-8 items-center justify-center rounded-lg",

   646	                    : t("codexAssistant.installCli")}
   647	                </span>
   648	              </span>
   649	              {!cliReady && (
   650	                <ChevronRight className="h-4 w-4 text-muted-foreground" />

   653	
   654	            <button
   655	              type="button"
   656	              onClick={() => {
   657	                closeAssistant();
   658	                onOpenCodexConfiguration();
   659	              }}
   660	              className="flex w-full items-center gap-3 rounded-xl border border-border bg-card px-3 py-3 text-left transition hover:border-violet-500/30"
   661	            >
   662	              <span
   663	                className={cn(
   664	                  "flex h-8 w-8 items-center justify-center rounded-lg",
   665	                  effectiveProviderReady
   666	                    ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-300"
   667	                    : "bg-amber-500/10 text-amber-600 dark:text-amber-300",
   668	                )}
   669	              >
   670	                {effectiveProviderReady ? (
   671	                  <Check className="h-4 w-4" />
   672	                ) : (
   673	                  <CircleAlert className="h-4 w-4" />
   674	                )}
   675	              </span>
   676	              <span className="min-w-0 flex-1">
   677	                <span className="block text-sm font-medium">
   678	                  {t("codexAssistant.modelProvider")}
   679	                </span>
   680	                <span className="block text-xs text-muted-foreground">
   681	                  {effectiveProviderReady
   682	                    ? t("codexAssistant.providerReady")
   683	                    : t("codexAssistant.configureProvider")}
   684	                </span>
   685	              </span>
   686	              <ChevronRight className="h-4 w-4 text-muted-foreground" />
   687	            </button>
   688	          </section>
   689	
   690	          <section className="space-y-3">
   691	            <p className="px-1 text-xs font-medium uppercase tracking-[0.12em] text-muted-foreground">
   692	              {t("codexAssistant.chatLabel")}
   693	            </p>
   694	            {webBridgeActive && (
   695	              <p className="rounded-lg border border-sky-500/20 bg-sky-500/[0.045] px-3 py-2 text-xs leading-relaxed text-sky-800 dark:text-sky-200">
   696	                {t("codexAssistant.webBridgeNotice")}
   697	              </p>
   698	            )}
   699	
   700	            <div className="space-y-2" aria-live="polite">
   701	              {chatMessages.length === 0 ? (

   703	                  {t("codexAssistant.chatEmpty")}
   704	                </p>
   705	              ) : (

   707	                  <div
   708	                    key={`${message.role}-${index}`}
   709	                    className={cn(
   710	                      "max-w-[92%] whitespace-pre-wrap break-words rounded-xl px-3 py-2 text-sm leading-relaxed",
   711	                      message.role === "user"
   712	                        ? "ml-auto bg-violet-500/15 text-foreground"
   713	                        : "mr-auto border border-border bg-card",
   714	                    )}
   715	                  >
   716	                    {message.content}
   717	                  </div>
   718	                ))
   719	              )}
   720	            </div>
   721	
   722	            {approvals.length > 0 && (
   723	              <div className="space-y-2" aria-live="assertive">
   724	                <p className="flex items-center gap-1.5 px-1 text-xs font-medium text-amber-700 dark:text-amber-200">
   725	                  <ShieldAlert className="h-3.5 w-3.5" />
   726	                  {t("codexAssistant.approvalRequired", {
   727	                    defaultValue: "Your approval is required",
   728	                  })}
   729	                </p>
   730	                {approvals.map((approval) => (
   731	                  <ApprovalCard
   732	                    key={approval.id}
   733	                    approval={approval}
   734	                    responding={respondingApprovalIds.has(approval.id)}
   735	                    onRespond={(decision) =>
   736	                      void respondToApproval(approval.id, decision)
   737	                    }
   738	                  />
   739	                ))}
   740	              </div>
   741	            )}
   742	
   743	            <div className="rounded-xl border border-border bg-card p-3">
   744	              <Input
   745	                value={request}
   746	                onChange={(event) => setRequest(event.target.value)}
   747	                onKeyDown={(event) => {
   748	                  if (event.key === "Enter") void sendMessage();
   749	                }}
   750	                disabled={!ready || running || startingSession}
   751	                placeholder={
   752	                  ready
   753	                    ? t("codexAssistant.chatPlaceholder")
   754	                    : (blockedReason ?? undefined)
   755	                }
   756	                className="border-0 bg-transparent px-0 shadow-none focus:ring-0"
   757	              />
   758	              <div className="mt-2 flex items-center justify-between border-t border-border/70 pt-2">
   759	                <span className="text-[11px] text-muted-foreground">
   760	                  {ready ? t("codexAssistant.chatHint") : blockedReason}
   761	                </span>
   762	                <Button
   763	                  size="icon"
   764	                  className="h-8 w-8 rounded-lg"
   765	                  disabled={
   766	                    !ready || !request.trim() || running || startingSession
   767	                  }
   768	                  onClick={() => void sendMessage()}
   769	                  aria-label={t("codexAssistant.sendChat")}
   770	                >
   771	                  {running || startingSession ? (
   772	                    <Loader2 className="h-3.5 w-3.5 animate-spin" />
   773	                  ) : (
   774	                    <Send className="h-3.5 w-3.5" />
   775	                  )}
   776	                </Button>
   777	              </div>
   778	            </div>
   779	
   780	            {running && sessionId && (
   781	              <Button
   782	                variant="outline"
   783	                size="sm"
   784	                className="w-full"
   785	                disabled={stopping}
   786	                onClick={() => void cancelRun()}
   787	              >
   788	                {stopping ? (
   789	                  <Loader2 className="h-3.5 w-3.5 animate-spin" />
   790	                ) : (
   791	                  <CircleStop className="h-3.5 w-3.5" />
   792	                )}
   793	                {stopping
   794	                  ? t("codexAssistant.cancelling")
   795	                  : t("codexAssistant.stopRun")}
   796	              </Button>
   797	            )}
   798	
   799	            {runError && (
   800	              <p className="rounded-lg border border-destructive/25 bg-destructive/5 px-3 py-2 text-xs leading-relaxed text-destructive">
   801	                {runError}
   802	              </p>
   803	            )}
   804	
   805	            {logs.length > 0 && (
   806	              <div className="overflow-hidden rounded-xl border border-border bg-muted/35">
   807	                <p className="border-b border-border px-3 py-2 text-[11px] font-medium uppercase tracking-[0.1em] text-muted-foreground">
   808	                  {t("codexAssistant.executionLog")}
   809	                </p>
   810	                <pre className="max-h-48 overflow-auto whitespace-pre-wrap break-words px-3 py-2 font-mono text-[11px] leading-relaxed text-muted-foreground">
   811	                  {logs.join("\n")}
   812	                </pre>
   813	              </div>
   814	            )}
   815	          </section>
   816	        </div>
   817	
   818	        <footer className="border-t border-border px-5 py-3 text-xs text-muted-foreground">
   819	          <span className="inline-flex items-center gap-1.5">
   820	            <MessageCircle className="h-3.5 w-3.5 text-violet-500" />
   821	            {t("codexAssistant.safetyNote")}
   822	          </span>
   823	        </footer>
   824	      </aside>
   825	    </>
   826	  );
