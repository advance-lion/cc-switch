   176	async function webAssistantRequest<T>(
   177	  path: string,
   178	  body?: Record<string, unknown>,
   179	): Promise<T> {
   180	  const response = await fetch(`${WEB_ASSISTANT_BRIDGE_BASE}${path}`, {
   181	    method: body ? "POST" : "GET",
   182	    headers: body ? { "Content-Type": "application/json" } : undefined,
   183	    body: body ? JSON.stringify(body) : undefined,
   184	  });
   185	  const payload = (await response.json()) as T & { error?: string };
   186	  if (!response.ok)
   187	    throw new Error(payload.error || "Web assistant bridge failed");
   188	  return payload;
   189	}
